##### Binary Search

1. 二分查找(精确查找)：

   ```cpp
   // arr must be sorted
   int BinarySearch(vector<int> &arr, int target) {
       if (arr.empty()) return false;
       // 设查找区间为 [left, right]
       int left = 0, right = arr.size() - 1;
       
       while (left <= right) { // 只要区间不为空
           int mid = left + (right - left) / 2;
   		
           if (arr[mid] == target) {
               return mid;
           } else if (arr[mid] < target) {
               left = mid + 1;
           } else {
               right = mid - 1;
           }
       }
       
       return -1; // 没找到
   }
   ```

2. 找 $\ge target$ 的第一个索引：

   ```cpp
   int FirstGreaterOrEqual(vector<int> &arr, int target) {
       int left = 0, right = arr.size() - 1; // 仍然是 [left, right]
       int ans = -1;
   
       while (left <= right) { // 只要区间不空
           int mid = left + (right - left) / 2;
           
           if (arr[mid] >= target) {
               ans = mid; // 记录可能解
               right = mid - 1;
           } else {
               left = mid + 1;
           }
       }
       
       return ans;
   }
   ```

3. 找 $\le target$ 的最后一个索引:

   ```cpp
   int LastLessOrEqual(vector<int>& arr, int target) {
       int left = 0, right = arr.size() - 1;
       int ans = -1;
   
       while (left <= right) {
           int mid = left + (right - left) / 2;
           
           if (arr[mid] <= target) {
               ans = mid; // 记录可行解
               left = mid + 1;
           }
           else {
               right = mid - 1;
           }
       }
       
       return ans;
   }
   ```

   > 推荐采在闭区间 $[left,right]$ 上搜索，不推荐 $[left,right)$ 这种半开半闭区间 (虽然C++标准库是后者)；
   >
   > 闭区间上，$left$ 和 $right$ 的更新逻辑都是 $mid\pm1$，不易错；

- 注：二分查找不一定必须在有序区间上；

  题目：[LeetCode - 162. 寻找峰值](https://leetcode.cn/problems/find-peak-element)；

  峰值元素：其值严格大于左右相邻值的元素；数组无序且无相等元素，可能包含多个峰值，返回任意一个即可；
  设 $arr[-1]=arr[n]=-\infty$；

  ```cpp
  int FindPeakElement(vector<int> &nums)
  {
      int n = nums.size();
      if (n == 1) return 0;
  
      // 单独验证两端是不是峰值
      if(nums[0] > nums[1]) return 0;
      if(nums[n-1] > nums[n-2]) return n - 1;
  
      // 两端都不是峰值，则左上升，右下降 ，且无相等，则中间必有峰值
      // 二分递归该过程即可
      int left = 1, right = n - 2;
      int ans = -1;
  
      while (left <= right)
      {
          int mid = left + ((right - left) >> 1);
  
          if (nums[mid-1] > nums[mid]) // 左区间必有峰值
              right = mid - 1;
          else if (nums[mid+1] > nums[mid]) // 右区间必有峰值
              left = mid + 1;
          else
          {
              ans = mid; // 中点恰好是峰值
              break;
          }
      }
  
      return ans;
  }
  ```

##### Problem 1

- 题目：[LeetCode - 875. 爱吃香蕉的珂珂](https://leetcode.cn/problems/koko-eating-bananas/)；

  简述一下题目：有 $n$ 堆香蕉，第 $i$ 堆有 $piles[i]$ 根；你每小时能吃某堆中的 $k$ 根，但如果这堆香蕉少于 $k$ 根，你只能将其吃完而不会多吃；问如果 $h$ 小时内你想吃完所有香蕉，$k$ 最小为？

  显然 $1\le k\le max\{piles[i]\}$；

  而且 $k$ 越大，吃完素有香蕉花费的小时数**必然不增**；

  所以可用二分查找的思想，找出花费小时数小于但是最接近 $h$ 的速度 $k$；

  ```cpp
  int minEatingSpeed(vector<int>& piles, int h) {
  	int low = 1, high = 0;
      for (int pile : piles) {
          high = max(high, pile);
      }
      
      int ans = 1;
  	while (low <= high) {
          int mid = low + (high - low) / 2;
          if (getHours(piles, mid) <= h) {
              // mid 满足条件，尝试寻找更小的可行速度
              ans = mid;
              high = mid - 1;
          } else {
              // mid 耗时超出 h，速度必须大于 mid
              low = mid + 1;
          }
      }
      
      return ans;
  }
  
  long long getHours(vector<int>& piles, int k) { // 这里用long long是因为实测发现int会越界
      long long hours = 0;
      for (int pile : piles) {
          hours += (pile + k - 1) / k; // 上取整计算单个堆耗时
      }
      return hours;
  }
  ```

##### Problem 2

- 题目：[LeetCode - 2560. 打家劫舍IV](https://leetcode.cn/problems/house-robber-iv)；

  题意简述：给定数组 $nums$，从中选择一个长度至少为 $k$ 的子序列 $A$，要求 $A$ 中没有任何元素在 $nums$ 中是相邻的；最小化 $max(A)$；(题目保证子序列 $A$ 存在) ($nums[i]\ge1$) ​

  思路：选择阈值 $X$，将满足 $nums[i]\le X$ 的元素选入子序列 $A$；如果在这样的子序列 $A$ 中能选出至少 $k$ 个在原数组中不相邻元素，说明 $X$ 可能取更小值；如果选不出来这样的 $k$ 个元素，那么必须增大 $X$；

  枚举 $min(nums)\le X\le max(nums)\implies$ 二分搜索；

  ```cpp
  int minCapability(vector<int>& nums, int k) {
  	int n = nums.size(), left = nums[0], right = nums[0];
      // 二分搜索范围
      for (int i = 0; i < n; i++) {
          left = min(left, nums[i]);
          right = max(right, nums[i]);
      }
      
      int ans = 0;
      while (left <= right) {
          int mid = left + (right - left) / 2;
          
          if (helper(nums, mid, k)) {
              ans = mid;
              right = mid - 1;
          } else {
              left = mid + 1;
          }
      }
      
      return ans;
  }
  
  // 主逻辑就是个二分搜索，看helper
  // 阈值为X时，能不能在nums[i]<=X的子序列中找出至少k个在原数组中不相邻的元素
  bool helper(const vector<int>& nums, int X, int k) {
      int n = nums.size();
      // dp[i] 表示前 i 个元素中最多能选出多少个互不相邻且 <= X 的数
      vector<int> dp(n + 1, 0);
  
      dp[0] = 0;
      dp[1] = (nums[0] <= X) ? 1 : 0;
  
      for (int i = 2; i <= n; i++) {
          if (nums[i - 1] <= X) {
              // 第i间房子(下标i-1)<=X时有两种选择，要么选要么不选
              dp[i] = max(dp[i - 1], dp[i - 2] + 1);
          } else {
              // 只能不选第i间房子
              dp[i] = dp[i - 1];
          }
  
          if (dp[i] >= k) return true; // 提前剪枝
      }
  
      return dp[n] >= k;
  }
  ```

  ```cpp
  // 空间压缩helper
  bool helper(const vector<int>& nums, int X, int k) {
      int prev2 = 0;                      // 相当于 dp[i - 2]
      int prev1 = (nums[0] <= X ? 1 : 0); // 相当于 dp[i - 1]
  
      if (prev1 >= k) return true;
  
      for (size_t i = 1; i < nums.size(); i++) {
          int cur = prev1; // 不选当前元素
          if (nums[i] <= X) {
              cur = max(cur, prev2 + 1); // 选当前元素
          }
          if (cur >= k) return true; // 提前剪枝
          
          prev2 = prev1;
          prev1 = cur;
      }
  
      return prev1 >= k;
  }
  ```

  ```cpp
  bool helper(const vector<int>& nums, int capability, int k) {
      int count = 0;
      int n = nums.size();
      for (int i = 0; i < n; ++i) {
          // 写法2：贪心选择，能选赶紧选
          if (nums[i] <= capability) {
              count++;
              i++; // 选了当前位置，相邻位置不可选，跳过下一个
              if (count >= k) {
                  return true;
              }
          }
      }
      return count >= k;
  }
  ```

  > $mid$ 最终能收敛到某个具体的元素 $nums[i]$；
