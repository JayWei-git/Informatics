# Map

- 图算法在得到某一顶点后常常需遍历其所有出边，直接去遍历边集是愚蠢的做法，所以需要建图：邻接矩阵 / 邻接表 / 链式前向星；
- 有时候如何将原问题建模成图才是最难的；

## Topological Sort

- 讲解：[左程云 - 算法059](https://www.bilibili.com/video/BV1rj411k7tS)；[左程云 - 算法060](https://www.bilibili.com/video/BV12y4y1F79q)；
- 拓扑排序：将有向无环图 (DAG) 所有顶点排成线性序列，使得 $\forall$ 有向边 $(u, v)$，序列中顶点 $u$ 都在 $v$ 之前；
  - 如果最终拓扑序列顶点个数 $<$ 图所有顶点数 $\implies$ 图中存在环；

- Kahn 算法：

  1. 统计所有顶点入度 并将初始入度为 $0$ 的顶点加入集合；
  2. 取出一个顶点 (任意) 并加入结果序列，并删除该顶点的 "影响"，若删去时某另一顶点入度减为 $0$，加入集合；
  3. 重复直到集合为空；若输出节点总数小于图的总节点数，说明图中有环；

  - Kahn 算法不在乎重复边，看起来重复边会导致顶点入度增加，不过由于每条边都会被遍历，其影响都会被删去；
  - 同一时刻集合中所有零入度节点等价，顶点出集合顺序是任意的，FIFO 队列，栈，堆，随机集合，甚至是逻辑上的集合都是可以的；
  - 时间复杂度为 $O(V + E)$，其中 $V$ 为顶点数，$E$ 为边数；

##### Problem 1

- 题目：[LeetCode - 210. 课程表II](https://leetcode.cn/problems/course-schedule-ii)；

  建模非常简单，不叙；
  
  ```cpp
  vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
      int n = numCourses, m = prerequisites.size();
      vector<int> nodes(n, -1), next(m, -1), to(m, -1);
      vector<int> in(n, 0);
  
      for (int j = 0; j < m; j++) { // 链式前向星建图并统计入度
          next[j] = nodes[prerequisites[j][1]]; // 这题的边是反着的
          nodes[prerequisites[j][1]] = j;
          to[j] = prerequisites[j][0];
          in[prerequisites[j][0]]++;
      }
  
      vector<int> ans(n); 
      int size = 0;
      int front = 0, rear = 0;
  
      for (int i = 0; i < n; i++) {
          if (in[i] == 0) {
              ans[rear++] = i;
              size++;
          }
      }
  
      while (front < rear) { // [front, rear) 就是上面说的零入度顶点集合
          int cur = ans[front++]; // 从集合中取出一个零入度顶点
  
          for (int j = nodes[cur]; j != -1; j = next[j]) { // 并删去其"影响"
              if (--in[to[j]] == 0) { // 发现新的零入度顶点
                  ans[rear++] = to[j]; // 加入集合
                  size++;
              }
          }
      }
  
      if (size != n) {
          return {}; // 有环
      } else {
          return ans;
      }
  }
  ```
  
  > 拓扑排序模板；

##### Problem 2

- [LeetCode - LCR 114. 火星词典](https://leetcode.cn/problems/Jf1JuT)；

  题意：改变字母的字典序 (比如 $a\to b\to c$ 变为 $b\to a \to c$)，然后按照这种字典序给出几个单词的字典序，要求还原出被打乱的字母字典序；

  建模思路：

  - 首先决定两个单词字典序的是首个不一样的字母，比如`abcazz`字典序在`abczaa`前面，说明存在边 $a\to z$，后面的字母无所谓，不能用来建图；

  - 只需要比较相邻单词，不需要两两比对，顺序具有传递性；
  
  字母字典序不存在的情况：一是拓扑排序后发现有环，二是给出的单词序列中存在后一个单词是前一个单词的前缀，比如`abcde`、`abc`；
  
  重复边不用理会，本节最开始也说了，Kahn算法不在乎；
  
  ```cpp
  string alienOrder(vector<string>& words) {
      // 1. 先统计出现了的字母
      vector<int> in(26, -1); // 入度为-1表示压根没出现
      for (const string& word : words) {
          for (char c : word) {
              in[c - 'a'] = 0;
          }
      }
      
      // 2. 建图并统计入度
      vector<vector<int>> graph(26); // 邻接表
      for (int i = 0; i < words.size() - 1; i++) {
          const string& word1 = words[i];
          const string& word2 = words[i + 1];
          
          int len = min(word1.size(), word2.size());
          int j = 0;
          for (; j < len; j++) {
              if (word1[j] != word2[j]) {
                  graph[word1[j] - 'a'].push_back(word2[j] - 'a'); // 只用首个不一致字母建图
                  in[word2[j] - 'a']++;
                  break; // 必须直接break，后续不能用于建图
              }
          }
          
          // 特判：如果word2是word1的前缀，那么不可能存在合法的字母字典序
          if (j < word1.size() && j == word2.size()) {
              return ""; // 直接return
          }
      }
      
      string ans = "";
      int n = 0; // 统计出现的字母个数
      int front = 0; // 这里不需要rear, ans.size()充当rear
  
      for (int i = 0; i < in.size(); i++) {
          if (in[i] != -1) { n++; }
          if (in[i] == 0) { ans.push_back((char)(i + 'a')); }
      }
      
      // 3. 拓扑排序
      while (front < ans.size()) {
          int cur = ans[front++] - 'a';
          
          for (int next : graph[cur]) {
              if (--in[next] == 0) {
                  ans.push_back((char)(next + 'a'));
              }
          }
      }
      
      return ans.size() == n ? ans : ""; // 判断有无环
  }
  ```

##### Problem 4 - Push-DP

- [洛谷 - P4017 最大食物链计数](https://www.luogu.com.cn/problem/P4017)；

    称从零入度顶点 (生产者) 开始的路径为`零起路径`；设 $dp[i]$ 表示终点为顶点 $i$ 的零起路径数量，显然 $ans = \sum dp[零出度顶点(最高消费者)]$；

    - 转移方程 (Pull视角)：$dp[i] = \sum dp[指向i的顶点]$  (拉取所有前驱信息)；
  - 转移方程 (Push视角)：当节点 $u$ 的状态确定 (出队时)，将其贡献直接累加给每一个后继节点 $v$：$dp[v] = (dp[v] + dp[u]) \pmod{MOD}$；
    - 初始状态：$dp[生产者]=1$；

    ```cpp
    #include <iostream>
    #include <vector>
    using namespace std;
    const int MOD = 80112002;
    
    int main() {
        int n, m;
        cin >> n >> m;
        vector<int> nodes(n, -1), next(m, -1), to(m, -1); // 链式前向星
        vector<int> in(n, 0), out(n, 0);
        for (int j = 0; j < m; j++) {
            int u, v;
            cin >> u >> v; // 注意题目顶点索引是1-based
            next[j] = nodes[u - 1];
            nodes[u - 1] = j;
            to[j] = v - 1;
    
            out[u - 1]++;
            in[v - 1]++;
        }
    
        vector<int> topo(n), dp(n, 0);
        int front = 0, rear = 0;
        for (int i = 0; i < n; i++) {
            if (in[i] == 0) {
                topo[rear++] = i;  // 生产者入队
                dp[i] = 1; // 初始状态
            }
        }
    
        while (front < rear) {
            int cur = topo[front++];
    
            for (int j = nodes[cur]; j != -1; j = next[j]) {
                if (--in[to[j]] == 0) {
                    topo[rear++] = to[j];
                }
    
                dp[to[j]] = (dp[cur] + dp[to[j]]) % MOD; // 不要放到上面的if里面，push是无条件的
            }
        }
    
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (out[i] == 0) { // 累加最高消费者
                ans = (ans + dp[i]) % MOD;
            }
        }   
        cout << ans << "\n";
        return 0;
    }

##### Problem 5

- [LeetCode - 851. 喧闹和富有](https://leetcode.cn/problems/loud-and-rich)；

  题意简述：要求返回一个`answer`数组，其中`answer[i]`为 不比第`i`号人穷的人中，`quiet`值最小那个人的编号；

  仍然是`Push-DP`；

  初始状态：所有人都认为自己最安静；
  
  题目将 $u\to v$ 设定为 $u$ 比 $v$ 富有也是便于拓扑时不停`Push`；
  
  ```cpp
  vector<int> loudAndRich(vector<vector<int>>& richer, vector<int>& quiet) {
      int n = quiet.size(), m = richer.size();
      vector<int> nodes(n, -1), to(m, -1), next(m, -1), in(n, 0);
      for (int j = 0; j < m; j++) {
          next[j] = nodes[richer[j][0]];
          nodes[richer[j][0]] = j;
          to[j] = richer[j][1];
          in[richer[j][1]]++;
      }
  
      vector<int> topo(n), dp(n);
      int front = 0, rear = 0;
      for (int i = 0; i < n; i++) {
          dp[i] = i;
          if (in[i] == 0) {
              topo[rear++] = i;
          }
      }
  
      while (front < rear) {
          int cur = topo[front++];
  
          for (int j = nodes[cur]; j != -1; j = next[j]) {
              int nxt = to[j];
              if (--in[nxt] == 0) {
                  topo[rear++] = nxt;
              }
  
              if (quiet[dp[nxt]] > quiet[dp[cur]]) { // Push
                  dp[nxt] = dp[cur];
              }
          }
      }
  
      return dp;
  }
  ```
  
  > 为什么初始值要设置为所有人都认为自己最安静，即`dp[i] = i`？
  >
  > 求解`dp[i]`的本质是 在顶点 $i$ 及其所有前驱顶点组成的集合中求最小值，`Push-DP`只是将这个过程变成了在拓扑排序的过程中两两比较而已；如果不将`dp[i]`设置为`i`，相当于漏掉了自身，在遇到所有比它更有钱的人，全都没它安静时就会出错；

##### Problem 6

- [LeetCode - 2050. 并行课程III](https://leetcode.cn/problems/parallel-courses-iii)；

  没什么好说的，和前两个题一样的`Push-DP`，设 $dp[i]$ 表示 修完课程 $i$ **之前**的所有课程需要花费的最小时间；

  ```cpp
  int minimumTime(int n, vector<vector<int>>& relations, vector<int>& time) {
      const int m = relations.size();
      vector<int> nodes(n, -1), to(m), next(m), in(n, 0);
      for (int j = 0; j < m; j++) {
          // relations是1-based编号，注意转为0-based编号
          to[j] = relations[j][1] - 1;
          next[j] = nodes[relations[j][0] - 1];
          nodes[relations[j][0] - 1] = j;
          in[relations[j][1] - 1]++;
      }
  
      vector<int> topo(n);
      vector<int> dp(n, 0); // 初始化为0，想想为什么？如果初始化dp[i] = time[i]，dp语义应该怎么改？
      int front = 0, rear = 0;
      int ans = 0;
  
      for (int i = 0; i < n; i++){
          if (in[i] == 0) {
              topo[rear++] = i;
          }
      }
  
      while (front < rear) {
          int cur = topo[front++];
          ans = max(ans, dp[cur] + time[cur]); // 所有课程完成时间的最大值即为总完成时间
  
          for (int j = nodes[cur]; j != -1; j = next[j]) {
              int nxt = to[j];
              if (--in[nxt] == 0) {
                  topo[rear++] = nxt;
              }
              
              dp[nxt] = max(dp[cur] + time[cur], dp[nxt]); // Push - DP
          }
      }
  
      return ans;
  }
  ```

##### Problem 8

- 题目：[LeetCode - 1203. 项目管理](https://leetcode.cn/problems/sort-items-by-groups-respecting-dependencies)；(Added by Myself)

  此题相较于简单地对项目进行拓扑排序，新增了一个要求：同一时刻只能存在一个小组在干活，所以某个小组只能等到前面所有小组把活干完才能轮到自己；
  
  本节开头就说过，Kahn 算法维护的集合可以使用任意结构，那么这里直接假设该集合每次吐出的项目是小组拓扑序最小的就行了；(相当于按小组拓扑序排序的堆)
  
  其实和什么最小拓扑序之类的差不多，集合吐顶点的逻辑不同罢了；
  
  ```cpp
  vector<int> sortItems(int n, int m, vector<int>& group, vector<vector<int>>& beforeItems) {
  	// 无组负责的项目假设由不同的新的小组负责，予以小组编号
  	int groupCount = m;
  	for (int& groupId : group) {
  		if (groupId == -1) {
  			groupId = groupCount++;
  		}
  	}
  
  	vector<vector<int>> groupGraph(groupCount);
  	vector<int> groupIndegree(groupCount, 0);
  
  	vector<vector<int>> itemGraph(n);
  	vector<int> itemIndegree(n, 0);
  
  	// 建图
  	for (int itemId = 0; itemId < n; itemId++) {
  		for (const auto& pre : beforeItems[itemId]) {
  			itemGraph[pre].push_back(itemId);
  			itemIndegree[itemId]++;
  			
  			// 注意组不要添加自环
  			if (group[pre] != group[itemId]) {
  				groupGraph[group[pre]].push_back(group[itemId]);
  				groupIndegree[group[itemId]]++;
  			}
  		}
  	}
  
  	// 首先获取小组拓扑排序结果作为后续项目拓扑排序依据
  	vector<int> topoSortedGroup(groupCount);
  	int front = 0, rear = 0;
  	for (int j = 0; j < groupCount; j++) {
  		if (groupIndegree[j] == 0) {
  			topoSortedGroup[rear++] = j;
  		}
  	}
  	while (front < rear) {
  		int cur = topoSortedGroup[front++];
  		for (const int& next : groupGraph[cur]) {
  			if (--groupIndegree[next] == 0) {
  				topoSortedGroup[rear++] = next;
  			}
  		}
  	}
  
  	if (rear != groupCount) { return {}; } // 组间存在环
  	
  	vector<int> groupRank(groupCount);
  	for (int j = 0; j < groupCount; j++) {
  		groupRank[topoSortedGroup[j]] = j;
  	}
  
  	// 对项目进行拓扑排序
  	// 堆元素为{组排名，项目编号}，这样组排名靠前的项目先出堆入ans
  	priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> q;
  	for (int item = 0; item < n; item++) {
  		if (itemIndegree[item] == 0) {
  			q.push({groupRank[group[item]], item});
  		}
  	}
  	vector<int> ans;
  	ans.reserve(n);
  	while (!q.empty()) {
  		int cur = q.top().second;
  		q.pop();
  		ans.push_back(cur);
  
  		for (const auto& next : itemGraph[cur]) {
  			if (--itemIndegree[next] == 0) {
  				q.push({groupRank[group[next]], next});
  			}
  		}
  	}
  
  	if (ans.size() != n) { return {}; } // 项目存在环
  	return ans;
  }
  ```

## Minimum Spanning Tree

- 讲解：[左程云 - 算法061](https://www.bilibili.com/video/BV1sK4y1F7LH)；

- 在连通加权**无向图**中，最小生成树 (MST) 是指用总权重最小的边将所有顶点连通且不产生任何环路的树形子图；
  - $V$ 个顶点，最小生成树一定 $V-1$ 条边，多一条有环，少一条不连通；

1. `Kruskal算法`：
   - **按权排序**：把图中所有边按权值从小到大排序；
   - **贪心挑边**：从最小的边开始依次检查，只要加入这条边不会与已选的边**形成环** (即加边的两端不在同一个连通分量中)，就选入生成树；否则丢弃；
   - **查并维护**：用**并查集**判断是否成环并合并集合，选满 $V - 1$ 条边即构成最小生成树；
   - 不需要建图 (不需要高效遍历顶点的邻边)；
2. `Prim算法`：
   - 将顶点划分为两个集合：已选集合 $S$ 和未选集合 $V - S$；
   - 任意选定一个起始顶点加入 $S$；
   - 每次在连接 $S$ 与 $V - S$ 的所有跨集合边中，选取**权值最小**的那条边，将其对应的对端顶点加入 $S$；
   - 重复步骤 3，直到所有顶点均被加入 $S$；
   - 与 Kruskal 不同，Prim 算法始终维持单个连通分量扩张，天然不会产生环；
   - 有优化版本 (见下)；

##### Problem 9 - MST via K/P

- [洛谷 - P3366 [模板] 最小生成树](https://www.luogu.com.cn/problem/P3366)；

1. 使用`K`算法：

   ```cpp
   class Edge {
   public:
       int u, v, w;
       bool operator<(const Edge& other) const { // 用于sort
           return w < other.w;
       }
   };
   
   class UniteFindSet {
   // 并查集实现省略
   // 注意顶点索引是1-based即可
   };
   
   int main() {  
       int N, M;
       if (!(cin >> N >> M)) return 0;
   	
       // 1. 录入边集并按权值升序排列，不需要建图
       vector<Edge> edges(M);
       for (int i = 0; i < M; i++) {
           cin >> edges[i].u >> edges[i].v >> edges[i].w;
       }
       sort(edges.begin(), edges.end());
       
       // 2. 初始化并查集
       UniteFindSet UFS(N);
       
       // 3. 贪心加边生成树
       int totalWeight = 0;
       int chosedEdges = 0;
       
       for (const auto& edge : edges) {
           if (UFS.unite(edge.u, edge.v)) { // 如果两个顶点不连通
               totalWeight += edge.w;
               chosedEdges++;
               if (chosedEdges == N - 1) break;
           }
       }
       
       // 4. 输出答案
       if (chosedEdges == N - 1) cout << totalWeight << "\n"; else cout << "orz\n";
   
       return 0;
   }
   ```

   > 时间复杂度 = 边排序 + 并查集初始化 + 遍历边集生树 = $O(m\times \log m)+O(n+m)$；

2. 使用`P`算法：

   ```cpp
   class Edge {
   public:
       int to, w;
       // 重载 < 运算符以实现小根堆：w 较小的优先级更高
       bool operator<(const Edge& other) const {
           return w > other.w;
       }
   };
   
   int main() {
       int N, M;
       if (!(cin >> N >> M)) return 0;
       
       // 建图
       vector<vector<Edge>> graph(N + 1);
       for (int j = 0; j < M; j++) {
           int u, v, w;
           cin >> u >> v >> w;
           // 注意是无向边
           graph[u].push_back({v, w});
           graph[v].push_back({u, w});
           
       }
       
       vector<bool> visitedNodesSet(N + 1, false);
       priority_queue<Edge> pq;
       
       // 从 1 号顶点出发：将 1 的所有邻边推入堆中
       visitedNodesSet[1] = true;
       for (const auto& edge : graph[1]) {
           pq.push(edge);
       }
       int chosedNodes = 1;
       int totalWeight = 0;
       
       // Prim 主循环
       while (!pq.empty() && chosedNodes < N) {
           Edge cur = pq.top();
           pq.pop();
           
           int u = cur.to;
           // 如果目标点已经在生成树集合中，跳过(避免成环)
           if (visitedNodesSet[u]) continue;
           
           visitedNodesSet[u] = true;
           totalWeight += cur.w;
           chosedNodes++;
           
           // 将新加入的点 u 的所有外向有效边推入堆
           for (const auto& next_edge : graph[u]) {
               if (!visitedNodesSet[next_edge.to]) {
                   pq.push(next_edge);
               }
           }
       }
       
       // 连通性判断并输出
       if (chosedNodes == N) cout << totalWeight << "\n"; else cout << "orz\n";
       
       return 0;
   }
   ```

   >  时间复杂度 = 建图 + 出入堆开销 = $O(m\times\log m)+O(n+m)$；

##### Problem 10 -  Enhance of Prim 不会

- `Prim`算法本质：将部分树视为一个整体，选取其最小出边 (且对端顶点不属于部分树)；

  $\to$ 开销大头：最小出边的选取；

  - 此前痛点：通过小根堆维护该整体的所有外向出边，导致堆规模与边数 $M$ 强正相关，产生大量无效入堆与出堆开销；

- 优化：不维护所有冗余出边，仅维护部分树整体到未选节点的**当前最短跨越边**；
  - **引入状态数组**：用`dis[v]`记录已生成树到未选节点 $v$ 的当前已知最短距离；
  - **松弛过滤 (Relaxation) **：新节点入树后，只有当新出边权值满足 $w < dis[v]$ 时才更新并推入小根堆；其余劣质边直接被拦截，从而将堆的有效规模严格限制在点数 $O(N)$ 级别；

- STL `priority_queue` 缺陷：不支持原生修改或删除特定元素；例如堆中已有边 `{to: 3, w: 10}`，下一轮发现更优边 `{to: 3, w: 3}` 时，无法直接更新旧记录，只能推入新副本；旧劣质边残留在堆中形成脏数据，出堆时必须依赖 `if (visited[u]) continue;` 兜底进行惰性清理；
  - 所以需要手写索引堆，实现原地修改机制：维护 `pos[u]` 记录顶点 $u$ 当前在堆数组中的物理下标；
  
  ```cpp
  class Edge {
      int to, w;
  };
  
  class IndexedMinHeap {
  private:
      vector<Edge> heap;
      vector<int> pos;
    	int size;
      
      void swap_edges(int i, int j) {
          swap(heap[i], heap[j]);
          pos[heap[i].to] = i;
          pos[heap[j].to] = j;
      }
      
      void sift_up(int i) {
          while (i > 0 && heap[i].w < heap[(i - 1) / 2].w) {
              swap_edges(i, i / 2);
              i /= 2;
          }
      }
      
      void sift_down(int i) {
          int left = i * 2 + 1;
          while (left < size) {
              int smallerChild = left;
              if (i * 2 + 2 < size && heap[i * 2 + 2].w < heap[left].w) {
                  smallerChild = i * 2 + 2;
              }
              
              if (heap[i].w > heap[smallerChild].w) {
                  swap_edges(i, smallerChild);
                  i = smallerChild;
              } else {
                  break;
              }
          }
      }
      
  public:
      IndexedMinHeap(int N) : heap(N), pos(N, -1), size(0) {}
      
      bool empty() { return size == 0; }
      bool contains(int i) { return pos[i] != -1; } // 部分树能不能抵达顶点i
      int get_weight(int to) const { return heap[pos[to]].w; }
      
      void push(Edge e) {
          int idx = size;
          heap[idx] = e;
          pos[e.to] = idx;
          size++;
          sift_up(idx);
      }
      
      Edge pop() {
          Edge root = heap[0];
          size--;
          if (size > 0) {
              heap[0] = heap[size];
              pos[heap[0].to] = 0;
              sift_down(0);
          }
          pos[root.to] = -1; // 标记移出堆
          return root;
      }
      
      void decrease_key(int to, int new_w) {
          int idx = pos[to];
          heap[idx].w = new_w;
          sift_up(idx);
      }
  };
  
  class Graph {
  private:
      
  };
  
  int main() {
      int N, M;
      if (!(cin >> N >> M)) return 0;
         
  }

##### Problem 12

- 题目：[LeetCode - 1697. 检查边长度限制的路径是否存在](https://leetcode.cn/problems/checking-existence-of-edge-length-limited-paths/)；

  将查询要求按边权限制升序排序 (要记录原始下标)，同时边集也按边权升序排序；

  双指针推进：指针一处理到查询`[p, q, limit]`后，指针二将所有`edge.weight < limit`的边加入并查集，然后直接查`find(q) == find(p)`；

  注：这称为**离线查询**：不实时交互，一次收集所有查询，随意修改次序；

  ```cpp
  class UniteFindSet {
  private:
      vector<int> parent, rank;
  public:
      UniteFindSet(int n) : parent(n), rank(n, 0) {
          for (int i = 0; i < n; i++) { parent[i] = i; }
      }
      int find(int x) { return parent[x] == x ? x : parent[x] = find(parent[x]); }
      bool unite(int x, int y) {
          int rootX = find(x), rootY = find(y);
          if (rootX == rootY) { return false; }
          if (rank[rootX] < rank[rootY]) { parent[rootX] = rootY; } 
          else if (rank[rootX] > rank[rootY]) { parent[rootY] = rootX; } 
          else { parent[rootX] = rootY; rank[rootY]++; }
          return true;
      }
  };
  
  vector<bool> distanceLimitedPathsExist(int n, vector<vector<int>>& edgeList, vector<vector<int>>& queries) {
      int m = edgeList.size(), k = queries.size();
  
      sort(edgeList.begin(), edgeList.end(), [](const auto& edge1, const auto& edge2) { return edge1[2] < edge2[2]; });
  
      vector<int> qid(k); // 只排序索引，不动原数组
      for(int i = 0; i < k; i++) { qid[i] = i; }
      sort(qid.begin(), qid.end(), [&](int i, int j) { return queries[i][2] < queries[j][2]; }); // 这里要仔细理解
  
      UniteFindSet UFS(n);
      vector<bool> ans(k);
      int edgeIdx = 0;  // 指针2
  
      for (int i : qid) { // 指针1逐个移动，直接用循环替代
          const auto& query = queries[i];
      	int node1 = query[0], node2 = query[1], limit = query[2];
          while (edgeIdx < m && edgeList[edgeIdx][2] < limit) { // 题意要求严格小于
              UFS.unite(edgeList[edgeIdx][0], edgeList[edgeIdx][1]); // 指针2扫过的边扔进并查集
              edgeIdx++;
          }
          ans[i] = (UFS.find(node1) == UFS.find(node2));
      }
  
      return ans;
  }
  ```

## Breadth-First Search

- 讲解：[左程云 - 算法062](https://www.bilibili.com/video/BV1Dw411w7P5)；[左程云 - 算法063](https://www.bilibili.com/video/BV11w411y79P)；

- 求无权图 / 等权图两点间最短路径可采用BFS；

  $\to$ BFS 首次到达某个顶点所经过的边数，严格等于从起点到该顶点的**最短路径长度 / 权值**；

- 从起始顶点出发，访问所有**从未被访问过的**邻接点，再由邻接点向外扩充访问更远的顶点；

  $\to$ 数据结构：队列 + 访问标记表 / 哈希表；

  执行步骤：

  1. 将起始顶点标记为已访问并入队；
  2. 当队列非空时，弹出队首顶点 $u$；
  3. 遍历 $u$ 的所有未被访问的邻接顶点 $v$：将 $v$ 标记为已访问，并压入队列；
  4. 重复上述过程直至队列为空；

  - 弹出可以是 单点弹出，也可以是整层弹出；

- 复杂度分析：

  - 时间：邻接表表示为 $O(V + E)$，邻接矩阵表示为 $O(V^2)$；
  - 空间：$O(V)$，用于保存队列状态与 visited 标记；

##### Problem 14

- 题目：[LeetCode - 1162. 地图分析](https://leetcode.cn/problems/as-far-from-land-as-possible)；

  建模：将所有陆地合起来视为超级源点；即求该源点到所有海洋点的最大距离，如果用BFS，那么最后一块被访问的海洋即为所求；
  
  ```cpp
  int maxDistance(vector<vector<int>>& grid) {
      int n = grid.size();
      queue<pair<int, int>> q;
  
      // 1. 将所有陆地入队
      for (int i = 0; i < n; ++i) {
          for (int j = 0; j < n; ++j) {
              if (grid[i][j] == 1) {
                  q.push({i, j});
              }
          }
      }
  
      // 全是陆地或全是海洋
      if (q.empty() || q.size() == n * n) {
          return -1;
      }
  
      static constexpr int dx[4] = {0, 0, -1, 1};
      static constexpr int dy[4] = {1, -1, 0, 0};
  
      pair<int, int> last = {-1, -1};
  
      // 2. BFS 遍历
      while (!q.empty()) {
          last = q.front();
          auto [x, y] = last;
          q.pop();
  
          for (int i = 0; i < 4; ++i) {
              int nx = x + dx[i];
              int ny = y + dy[i];
  
              // 遇到未访问海洋
              if (nx >= 0 && nx < n && ny >= 0 && ny < n && grid[nx][ny] == 0) {
                  grid[nx][ny] = grid[x][y] + 1; // 距离直接累加
                  q.push({nx, ny});
              }
          }
      }
  
      // 最后一个弹出的单元格距离减去初始陆地值，即为最大曼哈顿距离
      return grid[last.first][last.second] - 1;
  }
  ```

##### Problem 15

- 题目：[LeetCode - 691. 拼图贴纸](https://leetcode.cn/problems/stickers-to-spell-word)；

  状态：首先贴纸和 `target` 字符顺序不影响结果，可转为字母频次；相同频次对应一个状态；

  转移：使用一张贴纸，从剩余频次中扣除它提供的字母；每次转移的代价都是 1，目标是到达全零状态；

  $\to$ 在无权图上找两点间最短路径，用BFS；

  - 对于该题，假设贴纸数为 $n$，则每个顶点都有 $n$ 条出边，单纯BFS的时间复杂度不可接受，必须做剪枝设计；
    1. 状态**入队时**标记已访问，BFS 此后再遇到该状态，路径长度只会大于或等于已有路径，无需再次入队；
    2. 由于贴纸的使用也没有顺序，每层枚举贴纸可以只枚举包含某个特定字母的贴纸，即每个顶点不用遍历所有出边；

  	vector<array<int, 26>> stickersCount(stickers.size());
  	for (int i = 0; i < stickers.size(); i++) {
  		for (const char& c : stickers[i]) {
  			stickersCount[i][c - 'a']++;
  		}
  	}

  	sort(target.begin(), target.end());
  
      // 准备队列和哈希表
  	queue<string> q;
  	unordered_set<string> visited;
  
      // 初始状态
  	q.push(target);
  	visited.insert(target);
  
  	int level = 0;
  	while (!q.empty()) {
  		int size = q.size();
  		
  		while (size--) { // 一次处理一层
  			string cur = q.front();
  			q.pop();
  			char first = cur.front(); // 用cur的首字母作参考，没有该字母的贴纸直接跳过
  
  			for (int i = 0; i < stickers.size(); i++) {
  				if (stickersCount[i][first - 'a'] == 0) continue; // 跳过该贴纸
  				
  				string next = "";
  				auto count = stickersCount[i];
  				for (char c : cur) { // next的计算需要仔细理解
  					if (count[c - 'a'] > 0) count[c - 'a']--;
  					else next += c;
  				}
  
  				if (next.empty()) {
  					return level + 1; // 贴纸给target消耗完了，返回贴纸数量
  				}
  
  				if (!visited.count(next)) {
  					visited.insert(next);
  					q.push(next); // 未处理过的状态才压入队列
  				}
  			}
  		}
  
  		level++; // 下一层
  	}
  
  	return -1;
  }

##### Problem 16 - 01BFS

- 如果图的权值只有0 / 1，可用01BFS求两点间最短路径；其实就是简化的Dijkstra，用双端队列模拟对 需延长路径 / 顶点的排序过程；

- 题目：[LeetCode - 2290. 到达角落需要移除障碍物的最小数目](https://leetcode.cn/problems/minimum-obstacle-removal-to-reach-corner)；

  这种网格图不用显式建图；

  ```cpp
  const int INF = 1e9;
  int minimumObstacles(vector<vector<int>>& grid) {
      int m = grid.size(), n = grid[0].size();
      deque<pair<int, int>> dq;
      vector<vector<int>> dist(m, vector<int>(n, INF));
  
      dq.push_front({0, 0});
      dist[0][0] = 0;
  
      const vector<int> dx = {0, -1, 0, 1};
      const vector<int> dy = {1, 0, -1, 0};
  
      while (!dq.empty()) {
          auto [x, y] = dq.front();
          dq.pop_front();
  		
          if (x == m - 1 && y == n - 1) return dist[x][y]; // 遇到终点直接返回
          
          for (int i = 0; i < 4; i++) {
              int nx = x + dx[i], ny = y + dy[i];
              if (nx < 0 || nx > m - 1 || ny < 0 || ny > n - 1) continue;
  
              int nd = dist[x][y] + grid[nx][ny];
              if (nd >= dist[nx][ny]) continue;
  
              // dist变短了，更新并重新排序
              dist[nx][ny] = nd;
              if (grid[nx][ny] == 0) { 
                  dq.push_front({nx, ny});
              } else {
                  dq.push_back({nx, ny});
              }
          }
      }
  
      return dist[m - 1][n - 1];
  }
  ```
  
  > 讨论一下双端队列是怎么模拟对`dist`的排序的：
  >
  > 首先由于权值只有0和1，所以队列里面`dist`的长度只有`d`和`d+1`；
  >
  > 然后每次拿开头的，更新只可能是将某个`d+1`更新成`d` 或将某个`INF`更新成`d`或`d+1；`
  >
  > 将`d+1`更新成`d`，那个旧的`d+1`还是后面没有删掉，不过不影响正确性；
  >
  > 实在是看不惯旧记录，可以：
  >
  > ```cpp
  > // 入队
  > dq.push_front({nd, nx, ny});    // 权重为 0 时
  > // dq.push_back({nd, nx, ny});   // 权重为 1 时
  > 
  > // 出队
  > auto [d, x, y] = dq.front();
  > dq.pop_front();
  > if (d != dist[x][y]) continue; // 这份记录已经过时
  > ```
  
  - 一个小优化：使用一维`dist`数组来减少跨行访问的Cache Miss；
  
  ```cpp
  int minimumObstacles(vector<vector<int>>& grid) {
      int m = grid.size(), n = grid[0].size();
      int total = m * n;
  
      // 1. 使用一维连续内存，极大提升 CPU Cache 命中率
      vector<int> dist(total, 1e9);
      deque<int> dq; // 队列中仅存单一整数 index = x * n + y，减少 pair 内存开销
  
      dist[0] = 0;
      dq.push_front(0);
  
      const int dirs[4][2] = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
  
      while (!dq.empty()) {
          int curr = dq.front();
          dq.pop_front();
          if (curr == total - 1) return dist[curr];
  
          int x = curr / n, y = curr % n, d = dist[curr];
  
          for (const auto& dir : dirs) {
              int nx = x + dir[0];
              int ny = y + dir[1];
  
              if (nx >= 0 && nx < m && ny >= 0 && ny < n) {
                  int next_idx = nx * n + ny;
                  int weight = grid[nx][ny];
  
                  if (d + weight < dist[next_idx]) {
                      dist[next_idx] = d + weight;
                      if (weight == 0) {
                          dq.push_front(next_idx);
                      } else {
                          dq.push_back(next_idx);
                      }
                  }
              }
          }
      }
  
      return dist[total - 1];
  }
  ```

- 题目：[LeetCode - 1368. 使网格图至少有一条有效路径的最小代价](https://leetcode.cn/problems/minimum-cost-to-make-at-least-one-valid-path-in-a-grid)；

  如何建模为01图？假设格子中箭头往右指，相当于这个格子往上左下走的代价都是 $1$，往右走代价是 $0$；

  ```cpp
  const int INF = 1e9;
  int minCost(vector<vector<int>>& grid) {
      int m = grid.size(), n = grid[0].size();
      int total = m * n;
      vector<int> dist(total, INF);
      deque<int> dq;
  
      dist[0] = 0;
      dq.push_front(0);
  
      const int dirs[4][2] = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}}; // 注意与题目方向对应
  
      while (!dq.empty()) {
          int cur = dq.front();
          dq.pop_front();
          if (cur == total - 1) return dist[cur];
  
          int x = cur / n, y = cur % n, d = dist[cur];
  
          for (int i = 0; i < 4; i++) {
              int nx = x + dirs[i][0], ny = y + dirs[i][1];
              if (nx >= 0 && nx < m && ny >= 0 && ny < n) {
                  int next_idx = nx * n + ny;
                  int weight = (grid[x][y] == i + 1) ? 0 : 1; // 相同代价为0，不同代价为1 
  
                  if (d + weight < dist[next_idx]) {
                      dist[next_idx] = d + weight;
                      if (weight == 0) {
                          dq.push_front(next_idx);
                      } else {
                          dq.push_back(next_idx);
                      }
                  }
              }
          }
      }
  
      return dist[total - 1];
  }

##### Problem 18

- 题目：[LeetCode - 126. 单词接龙 II](https://leetcode.cn/problems/word-ladder-ii/)；

  建模：建图，将只差一个字母的单词用边连起来，然后运行BFS就行了；
  
  不要显式建图，复杂度太高；
  
  建反图，给每个单词维护一个`vector<string> parent`，然后DFS从`endWork`开始回溯法生成所有路径即可；
  
  ```cpp
  void dfs(const string& cur, const string& beginWord, vector<string>& path,
  		 const unordered_map<string, vector<string>>& parents, vector<vector<string>>& ans) {
  	if (cur == beginWord) { // 抵达beginWord，路径已形成
  		ans.emplace_back(path.rbegin(), path.rend()); // 记得把path扔进ans前要反一下
  		return;
  	}
  
      // 回溯法遍历反图所有路径
      // 反图只由BFS得到的最短路径构成
  	for (const string& parent : parents.at(cur)) { // 枚举反图的所有出边
  		path.push_back(parent);
  		dfs(parent, beginWord, path, parents, ans);
  		path.pop_back(); // 回溯
  	}
  }
  
  
  bool bfs(const string& beginWord, const string& endWord, const unordered_set<string>& words,
           unordered_map<string, vector<string>>& parents) {
  	unordered_map<string, int> dist;
  	queue<string> q;
  
  	dist[beginWord] = 0;
  	q.push(beginWord);
  
  	bool found = false;
  	while (!q.empty() && !found) {
  		int levelSize = q.size();
  		while (levelSize--) { // 按层BFS：防止漏掉多条相同长度的最短路径
              				  // 在某层中发现endWord后要把这层遍历完
  			string cur = q.front();
  			q.pop();
  
  			for (size_t i = 0; i < cur.size(); i++) {
  				for (char j = 'a'; j <= 'z'; j++) { // 不显式建图，而是枚举可能的出边，看图中有没有该出边
  					if (j == cur[i]) continue;
  					string next = cur;
  					next[i] = j;
  					if (!words.count(next)) continue; // 没有该出边，跳过
  					if (next == endWord) found = true; // 抵达endWord了，把这层遍历完不用继续往下走了
                      								   // 也可以不按层BFS，而是记录第一次遇见endWord的路径长，把该长度的路径走完即可
  					
  					int nextDist = dist[cur] + 1;
  					if (!dist.count(next)) {
  						dist[next] = nextDist;
  						parents[next].push_back(cur); // 只用最短路径去构建反图
  						q.push(next);
  					} else if (dist[next] == nextDist) {
  						// 首先dist[next]不可能大于nextDist，因为BFS首次遇到一定最短
  						// 其次dist[next]如果小于nextDist，也不用管，因为此时是更长路径
  						// 相等时说明找到另一条相同长度的最短路径
  						parents[next].push_back(cur);
  					}
  				}
  			}
  		}
  	}
      
  	return found;
  }
  
  vector<vector<string>> findLadders(string beginWord, string endWord, vector<string>& wordList) {
  	if (beginWord == endWord) return {{beginWord}};
  	unordered_set<string> words(wordList.begin(), wordList.end()); // 弄成哈希表方便不建图BFS找有没有某条边
  	if (!words.count(endWord)) return {};
  
  	unordered_map<string, vector<string>> parents; // (最短路径形成的)反图的邻接表
  	bool found = bfs(beginWord, endWord, words, parents);
  
  	// 根据parents还原路径即可
  	vector<vector<string>> ans;
  	if (!found) return ans;
  	vector<string> path{endWord}; // 反图，从endWord开始
  	dfs(endWord, beginWord, path, parents, ans);
  	return ans;
  }

##### Problem 19 - Bidirectional BFS

- 题目：[LeetCode - 127. 单词接龙](https://leetcode.cn/problems/word-ladder/)；

  常规BFS搜索空间太大，可采用双向BFS；
  
  ```cpp
  int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
      if (beginWord == endWord) return 1;
      unordered_set<string> words(wordList.begin(), wordList.end());
      if (!words.count(endWord)) return 0;
  
      unordered_set<string> smallLevel, bigLevel, nextLevel;
      smallLevel.insert(beginWord);
      bigLevel.insert(endWord);
      words.erase(beginWord);
      words.erase(endWord);
  
      for (int levels = 2; !smallLevel.empty(); levels++) {
          for (const string& cur : smallLevel) {
              for (size_t i = 0; i < cur.size(); i++) {
                  for (char j = 'a'; j <= 'z'; j++) {
                      if (j == cur[i]) continue;
                      string next = cur;
                      next[i] = j;
  
                      // 这两行if顺序不能反，这里words不是总词表，而是未访问词表
                      if (bigLevel.count(next)) return levels;
                      if (!words.count(next)) continue;
  
                      nextLevel.insert(next);
                  }
              }
          }
  
          for (const string& word : nextLevel) words.erase(word);
  
          if (nextLevel.size() > bigLevel.size()) {
              smallLevel = bigLevel;
              bigLevel = nextLevel;
          } else {
              smallLevel = nextLevel;
          }
  
          nextLevel.clear();
      }
  
      return 0;
  }
  ```
  
  > 这里重点解释为什么要从`words`里删除：
  >
  > 因为没有`dist`或`visited`数组，不知道访问到某个顶点时，是不是首次访问；
  >
  > (比如使用`dist`，如果访问到某个顶点发现此时的路径长比原先的`dist`值更大，说明不是首次访问，此时是一条更长的路径，直接跳过)
  >
  > 所以要把访问过的顶点从`words`中擦掉，让`words`同时承担 “在词典中” 与 “尚未访问”；
  >
  > 然后为什么是先判断相遇，再判断`words`？
  >
  > 因为当`next`在对侧的`bigLevel`中时，它早已从`words`中删除了；先判断`words`会直接`continue`；
  >
  > 因此顺序必须是：
  >
  > ```cpp
  > if (bigLevel.count(next)) return levels;
  > if (!words.count(next)) continue;
  > nextLevel.insert(next);
  > ```
  >
  > 最后提一句，从`smallLevel`延长是减少`next`的枚举量，可能能少延长点，但是也不一定；

## Dijkstra Algorithm

- `Dijkstra`：非负任意权值图求两点间最短路径；

  - 维护一个`dist`数组表示到各个顶点的距离，每次延长最短的 (用堆)，而且这个最短的一定是最优解，不断重复；
  - 注：由于普通堆 / 优先级队列不支持修改堆中已有元素，所以会存在旧记录；


##### Problem 20

- 题目：[LeetCode - 743. 网络延迟时间](https://leetcode.cn/problems/network-delay-time/)；

  ```cpp
  #include <vector>
  #include <algorithm>
  
  using namespace std;
  
  class IndexedMinHeap {
  private:
  	// heap[heapIndex] = node
  	// pos[node] = heapIndex
      // if (heapIndex == -1) throw_error("该顶点不在堆中")
  	const vector<int>& quotedDist;
  	vector<int> heap, pos;
  	int heapSize;
  
  	void heapSwap(int index1, int index2) {
  		swap(heap[index1], heap[index2]);
  		pos[heap[index1]] = index1;
  		pos[heap[index2]] = index2;
  	}
  
  	void shiftUp(int index) {
  		int parent = (index - 1) / 2;
  
  		while (quotedDist[heap[index]] < quotedDist[heap[parent]]) {
  			heapSwap(index, parent);
  			index = parent;
  			parent = (index - 1) / 2;
  		}
  	}
  
  	void shiftDown(int index) {
  		while (2 * index + 1 < heapSize) {
  			int left = 2 * index + 1;
  			int smallerChild = left;
  			int right = left + 1;
  
  			if (right < heapSize && quotedDist[heap[right]] < quotedDist[heap[left]]) {
  				smallerChild = right;
  			}
  			
  			if (quotedDist[heap[index]] > quotedDist[heap[smallerChild]]) {
  				heapSwap(index, smallerChild);
  			} else {
  				break;
  			}
  
  			index = smallerChild;
  		}
  	}
  
  public:
  	IndexedMinHeap(const vector<int>& dist) : quotedDist(dist), heap(dist.size()), pos(dist.size(), -1), heapSize(0) {}
  	bool empty() { return heapSize == 0; }
  	
  	void push(int node) {
  		if (pos[node] == -1) {
  			heap[heapSize] = node;
  			pos[node] = heapSize;
  			shiftUp(heapSize);
  			heapSize++;
  		} else {
  			shiftUp(pos[node]);
  		}
  	}
  
  	void pop() {
  		int removed = heap[0];
  		heapSwap(0, heapSize - 1);
  		--heapSize;
  		pos[removed] = -1;
  		if (heapSize > 0) shiftDown(0);
  	}
  
  	int top() {
  		return heap[0];
  	}
  };
  
  class Solution {
  public:
  int networkDelayTime(vector<vector<int>>& times, int n, int k) {
      int m = times.size();
      vector<int> nodes(n, -1), next(m, -1), to(m, -1), weight(m, 0);
  
      for (int j = 0; j < m; j++) {
          next[j] = nodes[times[j][0] - 1];
          nodes[times[j][0] - 1] = j;
          weight[j] = times[j][2];
          to[j] = times[j][1] - 1;
      }
  
      int ans = 0;
      vector<int> dist(n, 1e9);
  	IndexedMinHeap pq(dist);
  	vector<bool> visited(n, false);
  
      dist[k - 1] = 0;
      pq.push(k - 1);
  
      while (!pq.empty()) {
          int cur = pq.top();
  		visited[cur] = true;
          pq.pop();
          ans = max(ans, dist[cur]);
  
          for (int k = nodes[cur]; k != -1; k = next[k]) {
              if (dist[cur] + weight[k] < dist[to[k]]) {
                  dist[to[k]] = dist[cur] + weight[k];
                  pq.push(to[k]);
              }
          }
      }
  
      bool allReached = true;
      for (const bool& v : visited) {
          allReached = allReached && v;
      }
  
      if (!allReached) return -1;
      return ans;
  }
  };

  private:

  	vector<int> heap, pos;
  	int hsize;
  
  	void shiftUp(int i) {
  		int node = heap[i];
  		while (i > 0) {
  			int parent = (i - 1) / 2;
  			if (dist[node] < dist[heap[parent]]) {
  				heap[i] = heap[parent];
  				pos[heap[i]] = i;
  				i = parent;
  			} else {
  				break;
  			}
  		}
  		heap[i] = node;
  		pos[node] = i;
  	}

          int node = heap[i];
  		while (i * 2 + 1 < hsize) {
  			int left = i * 2 + 1;
  			int smaller = left;
  			int right = left + 1;
  			if (right < hsize && dist[heap[right]] < dist[heap[left]]) { smaller = right; }
  			if (dist[node] > dist[heap[smaller]]) { 
  				heap[i] = heap[smaller];
                  pos[heap[i]] = i;
  				i = smaller;
  			} else { 
  				break; 
  			}
  		}
          heap[i] = node;
          pos[node] = i;
  	}
  
  public:
  	IndexedMinHeap(const vector<int>& qdist) : dist(qdist), heap(qdist.size()), pos(qdist.size(), -1), hsize(0) {}
  	
  	bool empty() { return hsize == 0; }
  
  	void push(int node) { // 调用者保证只加新的或者改小
  		if (pos[node] != -1) {
  			shiftUp(pos[node]);
  		} else {
  			pos[node] = hsize;
  			heap[hsize] = node;
  			shiftUp(hsize);
  			hsize++;
  		}
  	}
  
  	void pop() { // 调用者保证不在堆空时调用
          int removed = heap[0];
          pos[removed] = -1;
          hsize--;
          if (hsize > 0) {
              heap[0] = heap[hsize];
              pos[heap[0]] = 0;
              shiftDown(0);
          }
  	}
  
  	int top() {
  		return heap[0];
  	}
  };

  		IndexedMinHeap h(w);
          w[0] = 0;
  		h.push(0);

  		while(!h.empty()) {
  			int cur = h.top();
  			h.pop();
              if (cur == total - 1) return w[cur];
  			int x = cur / m, y = cur % m;
  
  			for (const auto& dir : dirs) {
  				int nx = x + dir[0], ny = y + dir[1];
  				if (nx >= 0 && nx <= n - 1 && ny >= 0 && ny <= m - 1) {

  					if (neww < w[nxt]) {
  						w[nxt] = neww;
  						h.push(nxt);
  					}
  				}
  			}
  		}
  
  		return w[total - 1];
  	}

  // GPT可读性优化版
  class Solution {
  public:
      int shortestPathAllKeys(vector<string>& grid) {
          const int rows = grid.size();
          const int cols = grid[0].size();
  
          // 找到起点，并统计钥匙数量。
          int startRow = 0, startCol = 0, keyCount = 0;
          for (int row = 0; row < rows; ++row) {
              for (int col = 0; col < cols; ++col) {
                  if (grid[row][col] == '@') {
                      startRow = row;
                      startCol = col;
                  }
                  if (grid[row][col] >= 'a' && grid[row][col] <= 'f') {
                      ++keyCount;
                  }
              }
          }
  
          const int targetMask = (1 << keyCount) - 1;
          // visited[row][col][mask]：以 mask 中的钥匙到达该格子的状态是否已入队。
          vector<vector<vector<bool>>> visited(
              rows, vector<vector<bool>>(cols, vector<bool>(1 << keyCount, false)));
          queue<array<int, 3>> states; // {行, 列, 已持有钥匙的位掩码}
          const int directions[4][2] = {{0, 1}, {0, -1}, {-1, 0}, {1, 0}};
  
          states.push({startRow, startCol, 0});
          visited[startRow][startCol][0] = true;
  
          int steps = 0;
          while (!states.empty()) {
              // 当前层的状态都走了 steps 步。
              int layerSize = states.size();
              while (layerSize--) {
                  auto current = states.front();
                  states.pop();
                  if (current[2] == targetMask) return steps;
  
                  int row = current[0], col = current[1], mask = current[2];
                  for (const auto& direction : directions) {
                      int nextRow = row + direction[0];
                      int nextCol = col + direction[1];
                      if (nextRow >= 0 && nextRow <= rows - 1 &&
                          nextCol >= 0 && nextCol <= cols - 1 &&
                          !visited[nextRow][nextCol][mask] && grid[nextRow][nextCol] != '#') {
                          char cell = grid[nextRow][nextCol];
                          if (cell == '.' || cell == '@') {
                              visited[nextRow][nextCol][mask] = true;
                              states.push({nextRow, nextCol, mask});
                          } else if (cell >= 'a' && cell <= 'f') {
                              int nextMask = mask | (1 << (cell - 'a'));
                              if (!visited[nextRow][nextCol][nextMask]) {
                                  visited[nextRow][nextCol][nextMask] = true;
                                  states.push({nextRow, nextCol, nextMask});
                              }
                          } else if (cell >= 'A' && cell <= 'F') {
                              int keyBit = 1 << (cell - 'A');
                              if (mask & keyBit) {
                                  visited[nextRow][nextCol][mask] = true;
                                  states.push({nextRow, nextCol, mask});
                              }
                          }
                      }
                  }
              }
              ++steps;
          }
          return -1;
      }
  };

  		vector<int> nodes(n, -1), next(2 * m, -1), to(2 * m, -1), w(2 * m, 0);
  		// vector<int> minNeed(n, cnt + 1); // 便于后续Dijkstra松弛时剪枝
  
  		for (int j = 0; j < m; j++) {
  			int u = paths[j][0], v = paths[j][1], weight = paths[j][2];
  			int e1 = j, e2 = j + m;
  			
  			next[e1] = nodes[u];
  			nodes[u] = e1;
  			to[e1] = v;
  
  			next[e2] = nodes[v];
  			nodes[v] = e2;
  			to[e2] = u;
  
  			w[e1] = weight;
  			w[e2] = weight;
  		}
  
  		// Dijkstra实际运行在分层图上，共cnt + 1层
  		int layers = cnt + 1;
  		// dist索引 = 顶点编号 * layers + 剩余电量
  		vector<int> dist(n * layers, 1e9);
  		IndexedMinHeap h(dist);
  		dist[start * layers] = 0;
  		h.push(start * layers);
  
  		while (!h.empty()) {
  			int cur = h.top();
  			h.pop();
  			
  			int city = cur / layers, power = cur % layers;
  			int time = dist[cur];
  			if (city == end) return time;
  
  			// 松弛第一类出边：行驶边
  			for (int e = nodes[city]; e != -1; e = next[e]) {
  				if (power < w[e]) continue; // 电不够，这条边走不了
  
  				int next = to[e] * layers + (power - w[e]);
  				int ntime = time + w[e];
  
  				if (ntime < dist[next]) {
  					dist[next] = ntime;
  					h.push(next);
  				}
  			}
  
  			// 松弛第二类出边：充电边
              // 为什么只用松弛充一格电的充电边？观察LeetCode两次提交的时间变化和codex的回答，你会知道
  			if (power < cnt) {
  				int next = cur + 1;
  				int ntime = time + charge[city];
  
  				if (ntime < dist[next]) {
  					dist[next] = ntime;
  					h.push(next);
  				}
  			}	
  		}
  		return -1;
      }
  };

  class IndexedMinHeap {
  private:
  	const vector<int>& dist;
  	vector<int> heap, pos;
  	int hsize;
  
  	void shiftUp(int i) {
  		int node = heap[i];
  		while (i > 0) {
  			int parent = (i - 1) / 2;
  			if (dist[node] < dist[heap[parent]]) {
  				heap[i] = heap[parent];
  				pos[heap[i]] = i;
  				i = parent;
  			} else {
  				break;
  			}
  		}
  		heap[i] = node;
  		pos[node] = i;
  	}
  
  	void shiftDown(int i) {
          int node = heap[i];
  		while (i * 2 + 1 < hsize) {
  			int left = i * 2 + 1;
  			int smaller = left;
  			int right = left + 1;
  			if (right < hsize && dist[heap[right]] < dist[heap[left]]) { smaller = right; }
  			if (dist[node] > dist[heap[smaller]]) { 
  				heap[i] = heap[smaller];
                  pos[heap[i]] = i;
  				i = smaller;
  			} else { 
  				break; 
  			}
  		}
          heap[i] = node;
          pos[node] = i;
  	}
  
  public:
  	IndexedMinHeap(const vector<int>& qdist) : dist(qdist), heap(qdist.size()), pos(qdist.size(), -1), hsize(0) {}
  	
  	bool empty() { return hsize == 0; }
  
  	void push(int node) { // 调用者保证只加新的或者改小
  		if (pos[node] != -1) {
  			shiftUp(pos[node]);
  		} else {
  			pos[node] = hsize;

              heap[0] = heap[hsize];
              pos[heap[0]] = 0;
              shiftDown(0);
          }
  	}
  
  	int top() {
  		return heap[0];
  	}
