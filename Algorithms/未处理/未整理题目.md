##### LC5

- 题目：[LeetCode - 5. 最长回文子串](https://leetcode.cn/problems/longest-palindromic-substring)；

  设 $dp[i][j]$ 代表 $s[i...j]$ 是不是回文串；那么显然：
  $$
  dp[i][j]=\begin{cases}
  dp[i+1][j-1],& (s[i]==s[j])\\
  false,& (s[i]!=s[j])
  \end{cases}
  $$
  依赖于左下，而且由于 $j>=i$ ，所以只用填半张表；

  初始化： $i==j$ 时 $dp[i][j]$ 肯定是 $true$；然后 $j-i=1$ 和 $j-i=2$ 只看 $s[i]==s[j]$ 否就可初始化；

  所以整体来看，$s[i]==s[j]$ 可分两种情况；$s[i]!=s[j]$ 直接为 $false$； 

  ```cpp
  string longestPalindrome(string s) {
      int len = s.length();
      if (len < 2) return s;
  
      vector<vector<bool>> dp(len, vector<bool>(len, false));
      for (int i = 0; i < len; i++) { // i == j 时dp[i][j]为true
          dp[i][i] = true;
      }
  
      int maxLen = 1;
      int begin = 0;
  
      // 只填半张表，根据依赖关系选择从下面开始填
      for (int i = len - 2; i >= 0; i--) {
          for (int j = i + 1; j < len; j++) {
              if (s[i] == s[j]) { // s[i]==s[j]分两种情况
                  if (j - i < 3) {
                      dp[i][j] = true;
                  } else {
                      dp[i][j] = dp[i + 1][j - 1];
                  }
              } else { // s[i]!=s[j]直接为false
                  dp[i][j] = false;
              }
  
              if (dp[i][j] && (j - i + 1 > maxLen)) {
                  maxLen = j - i + 1;
                  begin = i;
              }
          }
      }
  
      return s.substr(begin, maxLen);
  }
  ```

##### LC62

- 题目：[LeetCode - 62. 不同路径](https://leetcode.cn/problems/unique-paths)；

  ```cpp
  // dp[i][j]表示从[0,0]到达[i,j]的路径数
  // dp[i][j] = dp[i - 1][j] + dp[i][j - 1]
  // 空间压缩
  int uniquePaths(int m, int n) {
      vector<int> dp(n, 1);
  
      for (int i = 1; i < m; i++) {
          for (int j = 1; j < n; j++) {
              dp[j] = dp[j] + dp[j - 1];
          }
      }
  
      return dp[n - 1];
  }
  ```

  > 第二层循环不要倒填 `for (int j = n - 1; j > 0; j--)`，这样是错的，因为行内有依赖关系，右依赖左，必须先填左；

  ```cpp
  int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
      int m = obstacleGrid.size();
      int n = obstacleGrid[0].size();
  
      // 1. 起点若为障碍物，直接返回 0 
      // 这里必须特判起点，因为下文会无条件赋值dp[0][0] = 1
      if (obstacleGrid[0][0] == 1) {
          return 0;
      }
  
      vector<vector<int>> dp(m, vector<int>(n, 0));
      dp[0][0] = 1;
  
      // 2. 初始化第一列：遇到障碍物立即终止
      for (int i = 1; i < m && obstacleGrid[i][0] == 0; i++) {
          dp[i][0] = 1;
      }
  
      // 3. 初始化第一行：遇到障碍物立即终止
      for (int j = 1; j < n && obstacleGrid[0][j] == 0; j++) {
          dp[0][j] = 1;
      }
  
      // 4. 状态转移：仅在无障碍时计算（dp 默认已为 0）
      for (int i = 1; i < m; i++) {
          for (int j = 1; j < n; j++) {
              if (obstacleGrid[i][j] == 0) {
                  dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
              }
          }
      }
  
      return dp[m - 1][n - 1];
  }
  ```

  ```cpp
  // 空间压缩
  int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
      int m = obstacleGrid.size();
      int n = obstacleGrid[0].size();
      
      if (obstacleGrid[0][0] == 1) {
          return 0;
      }
      
      vector<int> dp(n, 0);
      dp[0] = 1;
      for (int j = 1; j < n && obstacleGrid[0][j] == 0; j++) {
          dp[j] = 1;
      }
      
      for (int i = 1; i < m; i++) {
          // 填每一行之前特判第一个
          dp[0] = obstacleGrid[i][0] == 0 ? dp[0] : 0;
          for (int j = 1; j < n; j++) {
              if (obstacleGrid[i][j] == 0) {
                  dp[j] = dp[j] + dp[j - 1];
              } else {
                  // 遇到障碍物显式清零，否则是旧数据
                  dp[j] = 0;
              }
          }
      }
      
      return dp[n - 1];
  }
  ```

  > 注意空间压缩版中：
  >
  > 1. 对每行首元素的特判；
  > 2. 遇到障碍物要显式清除旧元素；

  ```cpp
  // dp[i]表示爬到第i + 1阶楼梯需要的方法数
  // dp[i] = dp[i - 1] + dp[i - 2]
  // 简单，没什么好说的
  int climbStairs(int n) {
      vector<int> dp(3, 0);
      dp[0] = 1;
      dp[1] = 2;
  
      for (int i = 2; i < n; i++) {
          dp[i % 3] = dp[(i - 1) % 3] + dp[(i - 2) % 3];
      }
  
      return dp[(n - 1) % 3];
  }
  ```

##### LC96

- 题目：[LeetCode - 96. 不同的二叉搜索树](https://leetcode.cn/problems/unique-binary-search-trees):

  二叉树的个数 $=$ 左子树个数 $\times$ 右子树个数；

  并且对于 BST ，其形态仅由**连续节点的数量**决定，与节点的具体数值无关；例如，由 $\{4, 5, 6\}$ 构成的 BST 数量与由 $\{1, 2, 3\}$ 构成的数量完全相同；

  设 $dp[len]$ 表示有 $len$ 个不同连续节点的 BST 数量；那么显然 $dp[0]=dp[1]=0$；

  枚举左子树节点数量，设为 $i$，那么对应右子树节点数量是 $len-i-1$，显然左右子树节点都是连续的，那么此时顶层 BST 数量为 $dp[i]\times dp[len-i-1]$；把所有 $i$ 对应结果累加即可；

  ```cpp
  int numTrees(int n) {
      vector<int> dp(n + 1, 0);
      dp[0] = 1;
      dp[1] = 1;
  
      for (int len = 2; len <= n; ++len) {
          for (int i = 0; i < len; ++i) { // 左子树节点数量从0到len-1
              dp[len] += dp[i] * dp[len - i - 1];
          }
      }
      return dp[n];
  }
  ```

##### LC207

- 题目：[LeetCode - 207. 课程表](https://leetcode.cn/problems/course-schedule/)；

  拓扑排序模板；

  ```cpp
  bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
      int n = numCourses, m = prerequisites.size();
      vector<int> nodes(n, -1), next(m), to(m), in(n, 0);
      for (int j = 0; j < m; j++) {
          next[j] = nodes[prerequisites[j][1]];
          nodes[prerequisites[j][1]] = j;
          to[j] = prerequisites[j][0];
          in[prerequisites[j][0]]++;
      }
  
      vector<int> topo(n);
      int front = 0, rear = 0;
      for (int i = 0; i < n; i++) {
          if (in[i] == 0) {
              topo[rear++] = i;
          }
      }
  
      while (front < rear) {
          int cur = topo[front++];
          for (int j = nodes[cur]; j != -1; j = next[j]) {
              if (--in[to[j]] == 0) {
                  topo[rear++] = to[j];
              }
          }
      }
  
      if (rear != n) {
          return false;
      } else {
          return true;
      }
  }
  ```

##### LC279

- 题目：[LeetCode - 279. 完全平方数](https://leetcode.cn/problems/perfect-squares)；

  设 $dp[i]$ 表示组成整数 $i$ 所需的最少完全平方数的个数；

  对于任意整数 $i$，遍历所有可能的完全平方数 $j^2$（其中 $1 \le j^2 \le i$）：
  $$
  dp[i] = \min_{1 \le j \le \lfloor\sqrt{i}\rfloor} \{ dp[i - j^2] + 1 \}
  $$

  ```cpp
  int numSquares(int n) {
      vector<int> dp(n + 1, INT_MAX);
      dp[0] = 0;
      
      for (int i = 1; i <= n; ++i) {
          // 逐个尝试每个完全平方数
          for (int j = 1; j * j <= i; ++j) {
              dp[i] = min(dp[i], dp[i - j * j] + 1);
          }
      }
      
      return dp[n];
  }
  ```

  > 让 $dp[i]$ 对应整数 $i$，不要对应数 $i+1$ 什么的，否则 $dp[i-j*j]$ 语义不清晰；
  >
  > $dp[0]$ 的作用：使递推能够自然地处理完全平方数本身；比如：$dp[4]=dp[4−4]+1=dp[0]+1=1$；
  >
  > - 此题实际为“完全背包模型”，物品重量依次为1、4、9、16.....物品价值全部为1，要让物品完全充满背包且总价值(相当于物品数)尽可能少；

##### LC322

- 题目：[LeetCode - 322. 零钱兑换](https://leetcode.cn/problems/coin-change)；

  和上面的 **279** 如出一辙；

  ```cpp
  int coinChange(vector<int>& coins, int amount) {
      vector<int> dp(amount + 1, amount + 1);
      dp[0] = 0;
  
      for (int i = 1; i <= amount; i++) {
          for (int j = 0; j < coins.size(); j++) {
              if (i >= coins[j]) {
                  dp[i] = min(dp[i], dp[i - coins[j]] + 1);
              }
          }
      }
  
      return dp[amount] <= amount ? dp[amount] : -1;
  }
  ```

  > - 相比 **279**，这里不知道 $i$ 和 $coins[j]$ 的大小关系，所以填表前需要判断：$if (i >= coins[j])$；
  >
  > - 但是这样有个问题，如果直接跳过这样的 $dp[i]$ 的更新，并且仍像 **279** 一样初始化为 $INT\_MAX$，那后面 $dp[i'-coins[j]] + 1(i'-coins[j]==i)$ 不就越界了？
  >
  > - 所以需要 $dp$ 数组初始化为 $amount + 1$，并且最后 `return dp[amount] <= amount ? dp[amount] : -1` 也满足了题意："如果没有任何一种硬币组合能组成总金额，返回 `-1`"；

##### LC407

- 题目：[LeetCode - 407. 接雨水 II](https://leetcode.cn/problems/trapping-rain-water-ii/)；

  将最外围柱子视为木桶，不断向内收缩木桶的边界；

  ```cpp
  struct Stave {
      int x, y, h;
      bool operator<(const Stave& other) const {
          return h > other.h; // 小根堆
      }
  };
  
  int trapRainWater(vector<vector<int>>& heightMap) {
      int m = heightMap.size();
      if (m <= 2) return 0;
      int n = heightMap[0].size();
      if (n <= 2) return 0;
      
  	priority_queue<Stave> pq;
      vector<vector<bool>> visited(m, vector<bool>(n, false));
      
      // 1. 将四周最外层的边界木板全部入堆
      for (int i = 0; i < m; i++) {
          pq.push({i, 0, heightMap[i][0]});
          visited[i][0] = true;
          pq.push({i, n - 1, heightMap[i][n - 1]});
          visited[i][n - 1] = true;
      }
      for (int j = 1; j < n - 1; j++) {
          pq.push({0, j, heightMap[0][j]});
          visited[0][j] = true;
          pq.push({m - 1, j, heightMap[m - 1][j]});
          visited[m - 1][j] = true;
      }
      
      int totalWater = 0;
      int dirs[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
      
      // 向内收缩木桶边界
      while (!pq.empty()) {
         Stave curr = pq.top();
          pq.pop();
          
          for (const auto& dir : dirs) {
              int nx = curr.x + dir[0], ny = curr.y + dir[1];
              
              if (nx >= 0 && nx < m && ny >= 0 && ny < n && !visited[nx][ny]) {
  				visited[nx][ny] = true;
                  
                  // 该格子高度小于木桶最短边，可积水
                  if (curr.h > heightMap[nx][ny]) {
                      totalWater += curr.h - heightMap[nx][ny];
                  }
                  
                  // 如果访问到的格子积水了，将积水后的高度作为木桶边界
                  // 没积水，直接将格子高度作为木桶边界
                  // 由此，木桶不断缩小，直到所有格子都被访问，木桶变为实心的
                  pq.push({nx, ny, max(curr.h, heightMap[nx][ny])});
              }
          }
      }
      
      return totalWater;
  }
  ```

  

##### LC1203

- 题目：[LeetCode - 1203. 项目管理](https://leetcode.cn/problems/sort-items-by-groups-respecting-dependencies)；

  此题相较于简单地对项目进行拓扑排序，新增了一个要求：同一时刻只能存在一个小组在干活，所以某个小组只能等到前面所有小组把活干完才能轮到自己；

  开篇就说过，Kahn 算法维护的集合可以使用任意结构，那么这里直接假设该集合每次吐出的项目是小组拓扑序最小的就行了；(相当于按小组拓扑序排序的堆)

  其实和什么最小拓扑序之类的差不多，集合吐顶点的逻辑不同罢了；

  ```cpp
  vector<int> sortItems(int n, int m, vector<int>& group, vector<vector<int>>& beforeItems) {
  	// 无组负责的项目假设由不同的新的小组负责，予以小组编号
  	int groupCount = m;
  	for (int& groupId : group) {
  		if (groupId == -1) {
  			groupId = groupCount++;
  		}
  	}
  
  	vector<vector<int>> groupGraph(groupCount);
  	vector<int> groupIndegree(groupCount, 0);
  
  	vector<vector<int>> itemGraph(n);
  	vector<int> itemIndegree(n, 0);
  
  	// 建图
  	for (int itemId = 0; itemId < n; itemId++) {
  		for (const auto& pre : beforeItems[itemId]) {
  			itemGraph[pre].push_back(itemId);
  			itemIndegree[itemId]++;
  			
  			// 注意组不要添加自环
  			if (group[pre] != group[itemId]) {
  				groupGraph[group[pre]].push_back(group[itemId]);
  				groupIndegree[group[itemId]]++;
  			}
  		}
  	}
  
  	// 首先获取小组拓扑排序结果作为后续项目拓扑排序依据
  	vector<int> topoSortedGroup(groupCount);
  	int front = 0, rear = 0;
  	for (int j = 0; j < groupCount; j++) {
  		if (groupIndegree[j] == 0) {
  			topoSortedGroup[rear++] = j;
  		}
  	}
  	while (front < rear) {
  		int cur = topoSortedGroup[front++];
  		for (const int& next : groupGraph[cur]) {
  			if (--groupIndegree[next] == 0) {
  				topoSortedGroup[rear++] = next;
  			}
  		}
  	}
  
  	if (rear != groupCount) { return {}; } // 组间存在环
  	
  	vector<int> groupRank(groupCount);
  	for (int j = 0; j < groupCount; j++) {
  		groupRank[topoSortedGroup[j]] = j;
  	}
  
  	// 对项目进行拓扑排序
  	// 堆元素为{组排名，项目编号}，这样组排名靠前的项目先出堆入ans
  	priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> q;
  	for (int item = 0; item < n; item++) {
  		if (itemIndegree[item] == 0) {
  			q.push({groupRank[group[item]], item});
  		}
  	}
  	vector<int> ans;
  	ans.reserve(n);
  	while (!q.empty()) {
  		int cur = q.top().second;
  		q.pop();
  		ans.push_back(cur);
  
  		for (const auto& next : itemGraph[cur]) {
  			if (--itemIndegree[next] == 0) {
  				q.push({groupRank[group[next]], next});
  			}
  		}
  	}
  
  	if (ans.size() != n) { return {}; } // 项目存在环
  	return ans;
  }
  ```
