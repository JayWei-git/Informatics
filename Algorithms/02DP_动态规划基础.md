##### Problem 1

- 题目：[LeetCode - 509. 斐波那契数](https://leetcode.cn/problems/fibonacci-number/description/)；

- 依赖关系很明确，$dp[i] = dp[i - 1] + dp[i - 2]$，填表即可，不用从递归逻辑出发；

  ```cpp
  int fib(int n) {
      if (n < 2) return n; // 当n=0时，下方dp[1]会越界
      
      std::vector<int> dp(n + 1, 0);
      dp[0] = 0;
      dp[1] = 1;
  
      for (int i = 2; i <= n; ++i) {
          dp[i] = dp[i - 1] + dp[i - 2];  // 状态转移方程
      }
  
      return dp[n];
  }
  ```

  ```cpp
  // 由于计算 dp[i] 时只需要用到前两个状态，因此只需保存前两个值即可
  int fib(int n) {
      if (n == 0) return 0;
      
      int prev2 = 0, prev1 = 1;
      
      for (int i = 2; i <= n; ++i) {
          int curr = prev1 + prev2;
          prev2 = prev1;
          prev1 = curr;
      }
      
      return prev1;
  }
  ```

##### Problem 2

- 题目：[LeetCode - 983. 最低票价](https://leetcode.cn/problems/minimum-cost-for-tickets/description/)；

- 递归逻辑：设 `int mincostAfter(int i)` 返回 `day[i]` ~  `days[n-1]` 的最小开销：

  (设 `days[] = {3,5,7,20,30,50}`) 那么 `mincostAfter(0)` 显然为下面三者中的最小值：

  1. `cost[0] + mincostAfter(1)`；因为只购买1天的通行证，`days[1]` 之后也就是第5天之后仍需购买通行证；
  2. `cost[1] + mincostAfter(3)`；购买了7天的通行证，第3天到第9天都不用买通信证了，`days[3]`也就是第20天之后仍需购买；
  3. `cost[2] + mincostAfter(5)`；原因同上；

  ```cpp
  vector<int> memory; // 记忆数组
  
  int mincostTickets(vector<int>& days, vector<int>& costs) {
      memory.assign(days.size(), -1); // -1表示未计算
      return mincostAfter(days, costs, 0);
  }
  
  int mincostAfter(const vector<int>& days, const vector<int>& costs, int i) {
      int n = days.size();
      if (i >= n) return 0;
      
      if (memory[i] != -1) return memory[i]; // 如果有记忆，直接返回
  
      // 期限为1天的通行证只能支撑1天
      int cost1 = costs[0] + mincostAfter(days, costs, i + 1);
      
      // 期限为7天的通信证可以管到 days[i] + 7 - 1 这一天
      // 所以需要寻找days中第一个 >= days[i] + 7 的日子的索引
      int next_7 = getNextIndex(days, i, days[i] + 7);
      int cost7 = costs[1] + mincostAfter(days, costs, next_7);
      
      // 同理
      int next_30 = getNextIndex(days, i, days[i] + 30);
      int cost30 = costs[2] + mincostAfter(days, costs, next_30);
  
      // 记录并返回三种决策中的最小值
      return memory[i] = min(cost1, min(cost7, cost30));
  }
  
  int getNextIndex(const vector<int>& days, int startIndex, int targetValue) {
  	int i = startIndex;
      // 寻找days[]中第一个 >= targetValue 的元素的索引
      // 如果 targetValue > days[n-1] 就返回 n
      // 是O(1)的，while最多循环30次
      while (i < days.size() && days[i] < targetValue) i++;
      return i;
  }
  ```

- 所以依赖关系为：
  $$
  dp[i] = \min \left\{
  \begin{aligned}
  & cost[0] + dp[gNI(days[i] + 1)], \\
  & cost[1] + dp[gNI(days[i] + 7)], \\
  & cost[2] + dp[gNI(days[i] + 30)]
  \end{aligned}
  \right\}(gNI(\ge days[n-1])=n)
  $$

  - 可知需要倒填表；

  ```cpp
  int mincostTickets(vector<int>& days, vector<int>& costs) {
      int n = days.size();
      vector<int> dp(n + 1, 0);
  
      for (int i = n - 1; i >= 0; --i) {
          int cost1 = costs[0] + dp[i + 1];
  
          int next7 = i;
          while (next7 < n && days[next7] < days[i] + 7) next7++;
          int cost2 = costs[1] + dp[next7];
  
          int next30 = i;
          while (next30 < n && days[next30] < days[i] + 30) next30++;
          int cost3 = costs[2] + dp[next30];
  
          dp[i] = min(cost1, min(cost2, cost3));
      } 
  
      return dp[0];
  }
  ```

  > 注意这里将 $dp$ 数组的大小声明为 $n+1$：将所有“一张票直接管到旅程结束”的越界情况，统一吸收为合法的 $+ dp[n]$ 计算，彻底消除了循环体内的越界特判 (哨兵节点)；

##### Problem 3

- 题目：[LeetCode - 91. 解码方法](https://leetcode.cn/problems/decode-ways/description/)；

- 递归逻辑：设 `decode(i)` 返回 `s[i]` 到 `s[n-1]` 的解码方法数；

  ```cpp
  int numDecodings(string s) {
      vector<int> memory(s.length(), -1);
      return decode(s, memory, 0);
  }
  
  int decode(const string& s,vector<int>& memory, int i) {
      int n = s.length();
      if (i == n) return 1; // 规定decode(n) = 1
      if (memory[i] != -1) return memory[i]; // Cache Hit
      if (s[i] == '0') return memory[i] = 0; // 前导零不可解码
      int ways = decode(s,memory, i + 1); // 先尝试单字解码
      if (i + 1 < n) { // 不是最后一个字符，就尝试双字解码
          int decodeNum = (s[i] - '0') * 10 + (s[i+1] - '0');
          if (decodeNum >= 10 && decodeNum <= 26) {
              ways += decode(s,memory, i + 2); // 加上双字解码
          }
      }
      return memory[i] = ways; // Cache Fresh
  }
  ```

  - 注意这个有个技巧是规定 `decode(n)=1`，可避免单独讨论解码末尾一两个字符；

- 所以依赖关系为：
  $$
  dp[i] = \begin{cases}  0, & \text{if } s[i] == '0' \\ dp[i+1] + (10 \le s[i..i+1] \le 26\text{ ? }dp[i+2] : 0), & \text{if } s[i] \neq '0' \end{cases}
  $$

  - 可知需倒填表

  ```cpp
  int numDecodings(string s) {
      int n = s.length();
      vector<int> dp(n + 1, 0);
      dp[n] = 1;
  
      for (int i = n - 1; i >= 0; i--) {
          if(s[i] == '0') {
              // s[i]为'0'，s[i]到s[n-1]没有解码方案
              dp[i] = 0;
              continue;
          }
  
          dp[i] = dp[i + 1]; // 尝试单字解码
  
          // 不是最后一个字符，就尝试双字解码
          if (i < n - 1) {
              int decodeNum = (s[i] - '0') * 10 + (s[i+1] - '0');
              if (decodeNum >= 10 && decodeNum <= 26) {
                  dp[i] += dp[i + 2];
              }
          }
      }
  
      return dp[0];
  }
  ```

  ```cpp
  // 空间压缩
  int numDecodings(string s) {
      int n = s.length();
      vector<int> dp(3, 1);
   	
      // dp[0]对应dp[i]，dp[1]对应dp[i+1]，dp[2]对应dp[i+2]
      // dp[1]初始化为1是因为规定了dp[n]=1
      // dp[0]和dp[2]初始化为什么都无所谓
      for (int i = n - 1; i >= 0; i--) {
          if(s[i] == '0') {
              dp[0] = 0; // 没有解码方案
          } else {
              dp[0] = dp[1]; // 单字解码
  
              // 不是最后一个字符，就尝试双字解码
              if (i < n - 1) {
                  int decodeNum = (s[i] - '0') * 10 + (s[i+1] - '0');
                  if (decodeNum >= 10 && decodeNum <= 26) {
                      dp[0] += dp[2];
                  }
              }
          }
          
          //  向后滚动
          dp[2] = dp[1];
          dp[1] = dp[0];
      }
      
      return dp[0];
  }
  ```

  > 此题和上一个题一样也使用了哨兵节点，不过值为 $1$ (想想为什么?)，所以哨兵节点的值是根据问题的具体定义来确定的；

##### Problem 4

- 题目：[LeetCode - 639. 解码方法Ⅱ](https://leetcode.cn/problems/decode-ways-ii/description/)；

- 和问题3差不多，递归逻辑就不写了；此题复杂在解码方式数量的讨论；

  ```cpp
  int numDecodings(string s) {
      int n = s.length();
      if (n == 0) return 0;
      
      long long M = 1000000007;
      vector<long long> dp(3, 1); // dp[0]:当前, dp[1]:i+1, dp[2]:i+2
      
      for (int i = n - 1; i >= 0; i--) {
          // 1. 单字解码
          long long decodeOne = (s[i] == '*') ? 9 : (s[i] != '0' ? 1 : 0);
          
          // 2. 双字解码
          long long decodeTwo = 0;
          if (i < n - 1) {
              // 开头不是*，1，2的双字解码直接免谈
              if (s[i] == '*') {
                  decodeTwo = (s[i + 1] == '*') ? 15 : ((s[i + 1] <= '6') ? 2 : 1);
              } else if (s[i] == '1') {
                  decodeTwo = (s[i + 1] == '*') ? 9 : 1;
              } else if (s[i] == '2') {
                  decodeTwo = (s[i + 1] == '*') ? 6 : ((s[i + 1] <= '6') ? 1 : 0);
              }
          }
          
          // 3. 计算并滚动
          dp[0] = (decodeOne * dp[1] + decodeTwo * dp[2]) % M; // 加法同余原理
          dp[2] = dp[1];
          dp[1] = dp[0];
      }
      
      return dp[0];
  }
  ```

##### Problem 5

- 题目：[LeetCode - 264. 丑数II](https://leetcode.cn/problems/ugly-number-ii/)；

  - 任何一个新丑数，必然是由“某个已有的丑数”乘以 2、3 或 5 得到的；

  - 所以依赖关系为：设数组 `dp[0...i-1]` 存放前 $i$ 个从小到大的丑数；则第 $i$ 个丑数 `dp[i]` 是：

    - $\mathcal{S}_2 = \{ dp[0]\times2, \ dp[1]\times2, \ dp[2]\times2, \ \dots \dots \}$

    - $\mathcal{S}_3 = \{ dp[0]\times3, \ dp[1]\times3, \ dp[2]\times3, \ \dots \dots \}$

    - $\mathcal{S}_5 = \{ dp[0]\times5, \ dp[1]\times5, \ dp[2]\times5, \ \dots \dots \}$

    这三个集合中大于 `dp[i-1]` 的所有数里面的最小值；

    但是如果每一轮都去遍历这三个集合，复杂度会演变成 $O(N^2)$；

  - 首先这三个集合中所有的元素都一定是丑数，所以所有元素都一定会被添加到序列中；其次如果某个结果 (比如 $2 \times 2 = 4$) 已经被挑中放进了序列，那么这个特定的“丑数 $\times$ 倍数”组合在下一轮就不用计算了；

  - 所以每个序列的排除顺序，一定是从左到右的连续排除；

  - 所以可以用三个指针 $p_2, p_3, p_5$ 分别代表下一轮要 $\times 2, \times 3, \times 5$ 的丑数，被添加了 指针就递增；

  ```cpp
  int nthUglyNumber(int n) {
  	vector<int> dp(n, 1);
      int p2 = 0, p3 = 0, p5 = 0;
  	
      for (int i = 1; i < n; i++) {
          dp[i] = min(dp[p2] * 2, min(dp[p3] * 3, dp[p5] * 5));
          
          // 思考：为什么不能使用 if-else，而是并列的if？
          // 比如2扩大3倍和3扩大2倍都是6，但是都要后移，if-else只能执行一个分支
          if (dp[i] == dp[p2] * 2) p2++;
          if (dp[i] == dp[p3] * 3) p3++;
          if (dp[i] == dp[p5] * 5) p5++;
      }
      
      return dp[n - 1];
  }
  ```

##### Problem 6

- 题目：[LeetCode - 32. 最长有效括号](https://leetcode.cn/problems/longest-valid-parentheses/description/)；注意 $(()()$ 答案是4而不是5；

  定义最长有效括号为一个Structure，比如 $()()()()(())((()))$；

  显然Structure的生成方式为：要么再套一个 $()$，要么与另一个Structure并列；

  所以依赖关系为：设 `dp[i]` 表示：从 `i` 位置向左最长Structure的长度；

  显然：

  - 如果 `s[i] == '('` ，`dp[i]` = 0；

  - 如果 `s[i] == ')'` ，且 `dp[i-1]` 对应的Structure的左边一个是 `'('`，那么 `dp[i] = dp[i-1] + 2 + dp[p]`；

    `dp[p]` 就是 `dp[i-1]` 对应的Structure的左边一个的 `'('`的左边一个的Structure；

    `+ 2`对应再套一层 ( )；`+dp[p]`对应并列一个Structure；

  显然是顺填表；

  ```cpp
  int longestValidParentheses(string s) {
      int n = s.length();
      if (n == 0) return 0; // 特判空字符串
  
      vector<int> dp(n, 0); // dp[0] = 0
      int ans = 0;
  
      for (int i = 1; i < n; i++) {
          if (s[i] == ')') { 
              int pre = i - dp[i - 1] - 1;
              
              if (pre >= 0 && s[pre] == '(') { // 越界检查
                  dp[i] = dp[i - 1] + 2; // 套一层()
                  
                  if (pre - 1 >= 0) { // 越界检查
                      dp[i] += dp[pre - 1]; // 并列一个Structure
                  }
              }
          }
          ans = max(ans, dp[i]); // 更新为最长
      }
      return ans;
  }
  ```

  > 将 $dp[i]$ 定义为向右最长 `Structure` 也是可行的，不过定义为向左更方便，比如：契合从左向右读取文本的数据流与前缀遍历逻辑、契合栈结构先进后出特性等；

##### Problem 7

- 题目：[LeetCode - 467. 环绕字符串中唯一的子字符串](https://leetcode.cn/problems/unique-substrings-in-wraparound-string/description/)；

  - 注意是不同的子字符串，比如 "zazazaza" 中子字符串 "za" 只统计一次；

- 设 `dp[i]` 为 $s[i]$ 结尾的最长连续环绕子串长度；

  则依赖关系为：
  $$
  L[i] = \begin{cases}    L[i-1] + 1, & \text{if } isNext(s[i-1], s[i]) \\   1, & \text{otherwise}   \end{cases}
  $$
  去重逻辑：同一个字母结尾的字符串只保留最长长度；

  ```cpp
  inline bool isNext(char pre, char cur) {
      return (cur - pre + 26) % 26 == 1; // 通过取模实现环状连续
  }
  
  int findSubstringInWraproundString(string s) {
      int n = s.length();
      if (n == 0) return 0; // 边界情况处理
  
      vector<int> dp(n, 1); // dp[0]=1
      vector<int> ans(26, 0); // 初始化为0，有的字母不一定出现
  
      // Base Case: s[0]
      ans[s[0] - 'a'] = 1;
  
      for (int i = 1; i < n; i++) {
          if (isNext(s[i - 1], s[i])) {
              dp[i] = dp[i - 1] + 1;
          } else {
              dp[i] = 1;
          }
          
          ans[s[i] - 'a'] = max(ans[s[i] - 'a'], dp[i]);
      }
      
      return accumulate(ans.begin(), ans.end(), 0); 
  }
  ```

  ```cpp
  // 由于只依赖前一个状态，可压缩空间
  int findSubstringInWraproundString(string s) {
      int n = s.length();
      if (n == 0) return 0;
      
      int dp = 1;// 旧dp就是dp[i-1]，新dp就是dp[i]，初始dp=dp[0]=1
      vector<int> ans(26, 0);
      
      ans[s[0] - 'a'] = 1;
     
      for (int i = 1; i < n; i++) {
          if (isNext(s[i - 1], s[i])) {
              dp++;
          } else {
              dp = 1;
          }
       
          ans[s[i] - 'a'] = max(ans[s[i] - 'a'], dp);
      }
      
      return accumulate(ans.begin(), ans.end(), 0);
  }
  ```

##### Problem 8

- 题目：[LeetCode - 940. 不同的子序列 II](https://leetcode.cn/problems/distinct-subsequences-ii/)；

- 如果设 `dp[i]` 是 `s[0...i]`产生的不同非空子序列数；

  在添加 `s[i]` 时，最自然的想法是：

  - 原来所有子序列都保留；
  - 原来所有子序列后面加上 `s[i]`；
  - 再加一个单独的 `s[i]`；

  即：`dp[i] = 2 * dp[i-1] + 1`；

  问题：会产生重复子序列；

- 如何去重？重复的核心原因是：新增字母 $\alpha$ 到 $s$ 末尾，必然会把当前所有子串中以 $\alpha$ 结尾的子串再次新增一次；

  可设 `end[c]` 表示当前所有子序列中以字符 c 结尾的子序列，则真正的依赖关系为：
  $$
  dp[i]=2*dp[i-1] + 1 - end[s[i]]
  $$
   其中 `end[s[i]]` 是添加 `s[i]` 之前的旧值；

- `end[]` 的更新逻辑为： 每添加字符 `s[i]`，新的以 `s[i]` 结尾的子序列共有：  `dp[i-1] + 1` 个：

  - 原先所有子序列后面接一个 `s[i]` 再加上 `s[i]` 本身；
  - 只要原先无重复子序列，那么尾部接一个相同的字符肯定也不会重复；

  ```cpp
  int distinctSubseqII(string s) {
      int n = s.length();
      vector<int> dp(n, 0);
      vector<int> end(26, 0);
      
      // 先单独处理Base Case，即s[0]
      dp[0] = 1;
      end[s[0] - 'a'] = 1;
      
      for (int i = 1; i < n; i++) {
          dp[i] = 2 * dp[i - 1] + 1 - end[s[i] - 'a'];
          end[s[i] - 'a'] = dp[i - 1] + 1;
      }
      
      return dp[n - 1];
  }
  ```

  ```cpp
  // 然后由于当前状态只依赖于前一个状态，所以可以压缩空间
  int distinctSubseqII(string s) {
      int dp;
      vector<int> end(26, 0);
  
      // 先单独处理Base Case，即s[0]
      dp = 1;
      end[s[0] - 'a'] = 1;
  
      for (int i = 1; i < s.length(); i++) {
          int temp = dp;
          dp = 2 * temp + 1 - end[s[i] - 'a'];
          end[s[i] - 'a'] = temp + 1;
      }
  
      return dp;
  }
  ```

  - 注意 LeetCode 要对结果取模；

##### Problem 9

- 题目：[LeetCode - 64. 最小路径和](https://leetcode.cn/problems/minimum-path-sum)；

1. 暴力递归做法：

   ```cpp
   // 设dfs(i,j)返回从grid[i][j]到grid[rows,cols]的最小路径和
   // 显然 dfs(i,j) = grid[i][j] + min(dfs(i + 1, j), dfs(i, j + 1))
   // 当前可引入一个temp_max用于剪枝，不过提升不大
   int dfs(vector<vector<int>>& grid, int i, int j) {
       int rows = grid.size(), cols = grid[0].size();
       if (i >= rows || j >= cols) return INT_MAX; // 越界返回无穷大，强制走另外一边
       if (i == rows - 1 && j == cols - 1) return grid[i][j]; // base case
       return grid[i][j] + min(dfs(grid, i + 1, j), dfs(grid, i, j + 1)); 
   }
   
   int minPathSum(vector<vector<int>>& grid) {
       return dfs(grid, 0, 0);
   }
   ```

   - dfs(i, j) 是会被重复计算的(不理解画个图)，而且递归树越往下重复计算次数是指数级增加的，可引入记忆化数组(略)；

2. 自底向上动态规划：

   设 `dp[i,j]` 值为从(0, 0)到(i, j)的最小路径和；(当然也可延续递归思路设为从(i, j)到右下角，不过这个依赖关系需要倒着填表)

   ```cpp
   // 转移方程：dp[i][j] = min(dp[i-1][j] + grid[i][j], dp[i][j-1] + grid[i][j])
   int minPathSum(vector<vector<int>>& grid) {
       int rows = grid.size(), cols = grid[0].size();
       vector<vector<int>> dp(rows, vector<int>(cols, 0));
   
       // 为避免处理越界，直接初始化第一行和第一列，for直接从i=1和j=1开始
       dp[0][0] = grid[0][0];
       for (int i = 1; i < rows; i++) {
           dp[i][0] = dp[i - 1][0] + grid[i][0];
       }
       for (int j = 1; j < cols; j++) {
           dp[0][j] = dp[0][j - 1] + grid[0][j];
       }
   
       for (int i = 1; i < rows; i++) {
           for (int j = 1; j < cols; j++) {
               dp[i][j] = min(dp[i - 1][j] + grid[i][j], dp[i][j - 1] + grid[i][j]);
           }
       }
   
       return dp[rows - 1][cols - 1];
   }
   
   ```

   - 动态规划时间复杂度 = 动态规划表大小 $\times$ 每个格子的枚举代价；

     所以这里是 $O(rows\times cols)$；

  ```cpp
      if (index == word.length()) { // 注意不是index == length - 1

      board[i][j] = '#';

                   dfs(board, i, j - 1, word, index + 1);

      board[i][j] = temp;
  
      return found;
  }
  ```

##### NC - 二叉树计数

- 题目：[NewCoder - 二叉树](https://www.nowcoder.com/practice/aaefe5896cce4204b276e213e725f3ea);

  递归函数定义为：$f(i,j)$ 值为 节点数为 $i$，高度不超过 $j$ 的二叉树方案数；

  消耗 1 个节点作为根节点，剩余 $i - 1$ 个节点分配给左子树和右子树；左右子树的高度均不能超过 $j - 1$：
  $$
  f(i, j) = \sum_{k=0}^{i-1} f(k, j-1) \times f(i - 1 - k, j-1) \pmod{10^9+7}
  $$
  Base Case: $f(0,j)=1$，$f(i,0)=0$，$f(0,0)=1$；(想想为什么空树返回1)

  然后不要疑惑会不会重复，因为二叉树左右子树是要区分的；

  ```cpp
  // 记忆化递归，按照递归逻辑和Base Case写就行了
  const int MOD = 1e9 + 7;
  
  long long dfs(int i, int j, vector<vector<long long>>& memo) {
  
      if (i == 0) return 1; // Base Case 1
      if (j == 0) return 0; // Base Case 2	
      if (memo[i][j] != -1) return memo[i][j]; // Cache hit
  
      long long res = 0;
      for (int k = 0; k < i; ++k) {
          long long left = dfs(k, j - 1, memo);
          long long right = dfs(i - 1 - k, j - 1, memo);
          res = (res + left * right) % MOD;
          // 这里绝不能写为 res += (left * right) % MOD; 每次加上一个小的数但最终结果会超过MOD，相当于没有取模
      }
  
      return memo[i][j] = res; // Cache fresh
  }
  
  int main() {
      int n, m;
      if (cin >> n >> m) {
          vector<vector<long long>> memo(n + 1, vector<long long>(m + 1, -1)); // 多一行是因为要到0
          cout << dfs(n, m, memo) << "\n";
      }
  
      return 0;
  }
  ```

##### <s>Problem 14</s>

- 题目：[LeetCode - 329. 矩阵中最长递增路径](https://leetcode.cn/problems/longest-increasing-path-in-a-matrix/description/)；

  这就是比较特殊的依赖关系，不好使用DP填表，那么记忆化递归就行了；

  ```cpp
  // 定义dfs(i, j)表示以坐标(i, j)为起始点能延伸出的最长递增路径长度
  class Solution {
  private:
      int rows, cols;
      const int dirs[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}}; // 四个方向：上、下、左、右
  
      int dfs(const vector<vector<int>>& matrix, int i, int j, vector<vector<int>>& memo) {
          if (memo[i][j] != 0) { return memo[i][j]; } // Cache Hit
  
          int maxLen = 1; // Base Case，四周都比自己小，那么路径长就是自己，为1
          
          for (auto& dir : dirs) {
              int ni = i + dir[0];
              int nj = j + dir[1];
  
              // 没有越界且有新的递增路径，才尝试更新maxLen
              if (ni >= 0 && ni < rows && nj >= 0 && nj < cols && matrix[ni][nj] > matrix[i][j]) {
                  maxLen = max(maxLen, 1 + dfs(matrix, ni, nj, memo)); // 仔细想想这个max
              }
          }
          
          return memo[i][j] = maxLen; // Cache Fresh
      }
  
  public:
      int longestIncreasingPath(vector<vector<int>>& matrix) {
          if (matrix.empty() || matrix[0].empty()) return 0;
          rows = matrix.size();
          cols = matrix[0].size();
          vector<vector<int>> memo(rows, vector<int>(cols, 0));
          int ans = 0;
  
          // 遍历每个格子作为起点
          for (int i = 0; i < rows; ++i) {
              for (int j = 0; j < cols; ++j) {
                  ans = max(ans, dfs(matrix, i, j, memo));
              }
          }
  
          return ans;
      }
  };
  ```

##### Problem 15

- 题目：[LeetCode - 115. 不同子序列](https://leetcode.cn/problems/distinct-subsequences)；

  从递归逻辑入手，还是从尾部字符相同还是不同来分类；

  设 $f(i,j)$ 是 `s[0...i]` 有多少个子序列等于 `t[0...j]`；

  - `s[i]!=t[j]`：那么 `t[j]` 不可能由 `s[i]` 匹配，只能由 `s[i]` 前面的去匹配 `t[j]` ，即：$f(i,j)=f(i-1,j)$；
  - `s[i]==t[j]`：分两种情况：一、让 `s[i]` 去匹配 `t[j]`，那么 $f(i,j)=f(i-1,j-1)$，二、`s[i]` 不匹配 `t[j]`，而是让 `s[i]` 之前的去匹配，那么 $f(i,j)=f(i-1,j)$；合二为一就是 $f(i,j)=f(i-1,j-1)+f(i-1,j)$；

  这样看填表顺序就是顺填；

  可以先单独初始化初始状态再填表；也可以多一行一列来容纳空串：$f(0,j)=0,f(i,0)=1,f(0,0)=1$；可举特例来定义空串行 / 列的值；

  ```cpp
  int numDistinct(string s, string t) {
      int n = s.length(), m = t.length();
      vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
      
      for (int i = 0; i <= n; i++) {
  		dp[i][0] = 1;
      }
  
      for (int i = 1; i <= n; i++) {
          for (int j = 1; j <= m; j++) {
              if (s[i - 1] == t[j - 1]) {
                  dp[i][j] = dp[i - 1][j - 1] + dp[i - 1][j];
              } else {
                  dp[i][j] = dp[i - 1][j];
              } 
          }
      }
  
      return dp[n][m];
  }
  ```

##### Problem 16

- 题目：[LeetCode - 72. 编辑距离](https://leetcode.cn/problems/edit-distance/)；

  还是从递归逻辑推依赖关系；

  仍然是两个字符串，所以直接定义 $f(i,j)$ 为将 `word1` 的前 $i$ 个字符转成 `word2` 的前 $j$ 个字符所需要的最少操作数；

  还是看尾部字符相不相同来分类讨论；

  - `word1[i]!=word2[j]`：

    1. 替换 `word1[i]` 为 `word2[j]`：$f(i,j)=f(i-1,j-1)+costOfReplace$；
    2. 删除 `word1[i]`：$f(i,j)=f(i-1,j)+costOfDelete$；
    3. 将 `word2[j]` 增加到 `word1[i+1]` 上：$f(i,j) = f(i,j-1)+costOfAdd$；

    三者取最小，即 `word1[i]!=word2[j]` 时：
    $$
    f(i,j) = \min \left\{
    \begin{aligned}
    & f(i-1,j-1)+costOfReplace, \\
    & f(i-1,j)+costOfDelete, \\
    & f(i,j-1)+costOfAdd
    \end{aligned}
    \right\}
    $$

  - `word1[i]==word2[j]`：$f(i,j)=f(i-1,j-1)$；

  ```cpp

  int minDistance(string word1, string word2) {
  	int n = word1.length(), m = word2.length();
      vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0)); // 仍然是多一行一列来容纳空串
      
      for (int j = 0; j <= m; j++) {
          dp[0][j] = j * cost[2]; // 将空串转为word2只能不断添加
      }
      
      for (int i = 0; i <= n; i++) {
  		dp[i][0] = i * cost[1]; // 将word1转为空串只能不断删除
      }
      
      for (int i = 1; i <= n; i++) {
          for (int j = 1; j <= m; j++) {
              if (word1[i - 1] != word2[j - 1]) {
                  dp[i][j] = min(
                  	dp[i - 1][j - 1] + cost[0],
                      min(dp[i - 1][j] + cost[1],

              }
          }
      }
      
      return dp[n][m];
  ```

  双指针的问题是只能选择一个分支 (移动两个指针之一)，那直接两个分支都递归；

  ```cpp
  class Solution {
      vector<vector<int>> memo;
  
      bool dfs(const string& s1, const string& s2, const string& s3, int p1, int p2) {
          int p3 = p1 + p2;
          if (p3 == s3.length()) return true;
  
          if (memo[p1][p2] != -1) return memo[p1][p2] == 1;
          
          bool match1 = (p1 < s1.length() && s1[p1] == s3[p3]) && dfs(s1, s2, s3, p1 + 1, p2);
          bool match2 = (p2 < s2.length() && s2[p2] == s3[p3]) && dfs(s1, s2, s3, p1, p2 + 1);
  
          memo[p1][p2] = (match1 || match2) ? 1 : 0; // 一个分支满足就行
          return memo[p1][p2] == 1;
      }
  
  public:
      bool isInterleave(string s1, string s2, string s3) {
          if (s1.length() + s2.length() != s3.length()) return false;
          memo = vector<vector<int>>(s1.length() + 1, vector<int>(s2.length() + 1, -1));
          return dfs(s1, s2, s3, 0, 0);
      }
  };
  ```

##### Problem 18

- 涂色问题：将 $n$ 个不同的格子填涂上 $m$ 种不同的颜色，要求每种颜色额至少填涂一次；共有多少种填涂方案？

  第二类斯特林数 $S(n,k)$，表示将含有 $n$ 个互不相同的元素的集合恰好划分为 $k$ 个非空子集的方案数；

  如果要把 $n$ 个元素划分为 $k$ 个非空子集：

  - 要么前面 $n-1$ 个元素已经被划分好为 $k$ 个集合，然后 $a_n$ 随便放进一个集合即可；

  - 要么前面 $n-1$ 个集合只被划分了 $k-1$ 个，$a_n$ 需要单独成为第 $k$ 个子集；
  - 只有这两种情况，前面 $n-1$ 个元素划分少于 $k-1$ 个子集的话不可能将 $n$ 个元素划分为 $k$ 个子集；

  $\implies S(n,k)=k\cdot S(n-1,k)+S(n-1,k-1)$；

- 现在有 $n$ 个不同的格子，先将它们划分为 $m$ 个不同集合，然后每个集合上一种颜色即可；

  若设 $dp[i][j]$ 为使用 $j$ 种颜色涂满 $i$ 个格子的有效方案数（即 $dp[i][j] = j! \cdot S(i, j)$，其动态规划转移方程为：
  $$
  dp[i][j] = dp[i-1][j-1] \cdot j + dp[i-1][j] \cdot j
  $$

- 边界条件：

  - $S(0, 0) = 1$
  - $S(n, 0) = 0 \quad (n > 0)$
  - $S(0, k) = 0 \quad (k > 0)$

  $dp[i][j]$ 依赖于左上和上，初始化第一行第一列，顺填；

  ```cpp
  long long countColorings2D(int n, int m, long long MOD = 1e9 + 7) {
      if (n < m || m <= 0) return (n == 0 && m == 0) ? 1 : 0;
      
      vector<vector<long long>> dp(n + 1, vector<long long>(m + 1, 0)); // 初始化第一行第一列
      dp[0][0] = 1; // 初始化 dp[0][0]
      
      for(int i = 1; i <= n; i++) {
          for (int j = 1; j <= min(i, m); j++) { // 只需要计算j<=i的格子，dp[i][j](j>i)肯定为0
              dp[i][j] = (j * (dp[i - 1][j - 1] + dp[i - 1][j])) % MOD; // 记得提取公因式减少乘法次数
          }
      }
      
      return dp[n][m];
  }
  ```

  ```cpp

      }
      
      return dp[m];
  }
  ```

##### Problem 19

- 给定两个字符串 $s1$ 和 $s2$，问 $s1$ 至少删掉多少个字符就成为 $s2$ 的子串；再次提醒：子串与子序列的区别；

  此题转移方程的得出灵感偏多，这里直接给出：设 $dp[i][j]$ 表示： $s1$ 的前 $i$ 个字符变成 $s2[l...l+j-1]$ 最少删除多少字符 ($l$ 任意，其实就是 $s2$ 前 $j$ 个字符的任意后缀串)；

  在这个定义下，最终需要返回最后一行中的最小值 (想想为什么？)；

  1. $s1[i-1]!=s2[j-1]$：只能删除 $s1[i-1]$，则 $dp[i][j]=dp[i-1][j]+1$；
  2. $s1[i-1]==s2[j-1]$：保留 $s1[i-1]$，则 $dp[i][j]=dp[i-1][j-1]$；

  ```cpp
  int minDelete(const string& s1, const string& s2) {
      int n = s1.length(), m = s2.length();
      
      vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
  
      for (int i = 1; i <= n; ++i) {
          dp[i][0] = i;
          for (int j = 1; j <= m; ++j) {
              if (s1[i - 1] == s2[j - 1]) {
                  dp[i][j] = dp[i - 1][j - 1];
              } else {
                  dp[i][j] = dp[i - 1][j] + 1;
              }
          }
      }
  
      int ans = INT_MAX;
      for (int j = 0; j <= m; ++j) {
          ans = min(ans, dp[n][j]);
      }
  
      return ans;
  }
  ```

- 题目：[LeetCode - 474. 一和零](https://leetcode.cn/problems/ones-and-zeroes)；

  多维费用背包问题；首先看递归方法，就是二分深搜，选或者不选当前字符串；

  ```cpp
  // 定义dfs(index, remainZeros, remainOnes)返回在strs[index]往后的元素中挑选满足remainZeros和remainOnes的最大元素个数
  // 这里也可看出为何不需要回溯，因为不用维护状态(只返回选了几个，不返回具体选了哪几个)
  int dfs(int index, int remainZeros, int remainOnes, 
          const vector<pair<int, int>>& counts, 
          vector<vector<vector<int>>>& memo) {
      // 边界条件
      if (index >= counts.size()) {
          return 0;
      }
  
      // 查备忘录
      if (memo[index][remainZeros][remainOnes] != -1) {
          return memo[index][remainZeros][remainOnes];
      }
  
      int zeros = counts[index].first;
      int ones = counts[index].second;
  
      // 分支 1：不选当前字符串
      int notTake = dfs(index + 1, remainZeros, remainOnes, counts, memo);
  
      // 分支 2：在容量允许的情况下选择当前字符串
      int take = 0;
      if (remainZeros >= zeros && remainOnes >= ones) {
          take = dfs(index + 1, remainZeros - zeros, remainOnes - ones, counts, memo) + 1;
      }
  
      return memo[index][remainZeros][remainOnes] = max(notTake, take);
  }

      // 预先统计 0 和 1 的数量
      for (int i = 0; i < len; i++) {
          int zeros = 0, ones = 0;
          for (char c : strs[i]) {
              if (c == '0') zeros++;
              else ones++;
          }
          counts[i] = {zeros, ones};
      }
  
      // 初始化三维 vector: 大小为 len * (m + 1) * (n + 1)，默认值 -1
      // 为什么第二维和第三维要 +1？ 想想
      vector<vector<vector<int>>> memo(
  ```

  然后来看动态规划怎么写：

  如果采用与递归dfs相同的语义，$dp[k][i][j]$ 表示在 $strs[k ... len-1]$ 中挑选，剩余容量为 $i$ 个 0 和 $j$ 个 1 时的最大数量；

  则递推公式为：
  $$
  dp[k][i][j] = \max(dp[k + 1][i][j], dp[k + 1][i - zeros][j - ones] + 1)
  $$
  由于 $zeros$ 和 $ones$ 是不确定的，无法说 $dp[k][i][j]$ 具体依赖于 $k+1$ 平面上的哪两个格子，只能说 $dp[k][i][j]$ 依赖于 $k+1$ 平面；

  因此再填下一个平面时，本层必须全部填完；

  由于本层之间无互相依赖，所以行优先还是列优先还是什么别的顺序都无所谓；

  这里也可窥见，由于只依赖于前一层，那么空间是可以压缩至二维的；

  再次提醒，选择分支要判断能不能选；

  ```cpp
  int findMaxForm(vector<string>& strs, int m, int n) {
      int len = strs.size();

      vector<vector<vector<int>>> dp(len + 1, vector<vector<int>>(m + 1, vector<int>(n + 1, 0)));
      
      for (int k = len - 1; k >= 0; k--) {
          int zeros = 0, ones = 0;
          for (char c : strs[k]) {
              if (c == '0') zeros++;
              else ones++;
          }

                      dp[k][i][j] = max(dp[k][i][j], dp[k + 1][i - zeros][j - ones] + 1);
                  }
              }
          }
      }
  ```

  也是多维费用背包问题；

  设 $dp[i][j][k]$ 为：前 $i$ 种工作中，能总共产生至少 $j$ 利润，且总共需要人数不超过 $k$ 的工作子集的个数；

     dp[i][j][k]=dp[i-1][max(0,j-profit[i])][k-group[i]]
     $$

##### Problem 23

- 题目：[LeetCode - 2435. 矩阵中和能被 K 整除的路径](https://leetcode.cn/problems/paths-in-matrix-whose-sum-is-divisible-by-k/)；

  ```cpp
  // 递归
  class Solution {
  private:
      const int MOD = 1e9 + 7;
      int mod_k;
      int m;
      int n;
      
      int dfs(vector<vector<int>>& grid, int remainder, int row, int column, 
              vector<vector<vector<int>>>& memo) {
          remainder = (remainder + grid[row][column]) % mod_k; // 同余原理，逐个相加取余等于全部相加取余
  
          // 如果到达终点，看余数是不是为0
          if (row == m - 1 && column == n - 1) {
              return (remainder == 0 ? 1 : 0);
          }
  
          if (memo[row][column][remainder] != -1) { // 查Cache
              return memo[row][column][remainder];
          }
          
          int paths = 0;
          if (row + 1 < m) { // 向下走
              paths = (paths + dfs(grid, remainder, row + 1, column, memo)) % MOD; 
          }
  
          if (column + 1 < n) { // 向右走
              paths = (paths + dfs(grid, remainder, row, column + 1, memo)) % MOD;
          }
  
          return memo[row][column][remainder] = paths;
      }
      
  public:
      int numberOfPaths(vector<vector<int>>& grid, int k) {
          // 题目保证矩阵至少一行一列，且矩阵元素非负
          mod_k = k;
          m = grid.size();
          n = grid[0].size();
          // 为什么记忆数组是三维的？ 因为dfs有三个状态变量: remainder, row, column
          vector<vector<vector<int>>> memo(m, vector<vector<int>>(n, vector<int>(k, -1)));
          return dfs(grid, 0, 0, 0, memo);
      }
  };
  ```

  所以：
  $$
  \begin{aligned} dfs(rem, r, c) &= \Big(\mathbb{I}(r+1 < m) \cdot dfs(newRem, r+1, c) + \mathbb{I}(c+1 < n) \cdot dfs(newRem, r, c+1)\Big) \bmod (10^9+7)  \\
   newRem &= (rem + grid[r][c]) \bmod k \end{aligned}
  $$
  (其中 $\mathbb{I}(\cdot)$ 为指示函数，条件不满足时项为 0)；

  所以可以想象 $row$ 为 $x$ 轴，$column$ 为 $y$ 轴，$remainder$ 为 $z$ 轴；

  仅就 $xy$ 平面而言，每个格子依赖于右和下 (没有右就仅依赖下，没有下就仅依赖右)，只是不知道被依赖格子的高度；

  因此，$\text{base case}$ 就是 $x=m-1,y=n-1$ 的整根 $z$ 轴柱子；

  ```cpp
  int numberOfPaths(vector<vector<int>>& grid, int k) {
      const int MOD = 1e9 + 7;
      int m = grid.size(), n = grid[0].size();
      // 余数值域为0~k-1，因此z轴高度为k
  	vector<vector<vector<int>>> dp(m, vector<vector<int>>(n, vector<int>(k, -1)));
      
      // 初始化base case
      // 柱子的高度就是前面路径和模k的余数
      for (int z = 0; z < k; z++) {
          if ((z + grid[m - 1][n - 1]) % k == 0) {
              dp[m - 1][n - 1][z] = 1;
          } else {
              dp[m - 1][n - 1][z] = 0;
          }
      }
      
      // 倒填表
      // 柱子的高度就是前面路径和模k的余数
      for (int r = m - 1; r >= 0; r--) {
          for (int c = n - 1; c >= 0; c--) {
              if (r == m - 1 && c == n - 1) continue; // 跳过base case
              for (int z = 0; z < k; z++) {
                  int paths = 0, newRem = (z + grid[r][c]) % k;
                  if (r + 1 < m) {
                      paths = (paths + dp[r + 1][c][newRem]) % MOD;
                  }
                 	if (c + 1 < n) {
                      paths = (paths + dp[r][c + 1][newRem]) % MOD;
                  }
                  
                  dp[r][c][z] = paths;
              }
          }
      }
      
      return dp[0][0][0];
  }
  ```

- 题目：[LeetCode - 87. 扰乱字符串](https://leetcode.cn/problems/scramble-string)；

  无论后续怎么翻转，某一步切出来的左半段字符集合，最终在目标串 $s_2$ 中也必须是**紧密连在一起的一整块连续子串**，绝对不可能有右半段的字符插到中间去；

  这提供了关键的分治思路：

  ```cpp
  // dfs返回s1从i位置开始，s2从j位置开始，长度为len的子字符串是否互为扰乱字符串
  bool dfs(const string& s1, const string& s2, int i, int j, int len, vector<vector<vector<int>>>& memo) { 
      // 查Cache
      if (memo[i][j][len] != 0) {
          return memo[i][j][len] == 1;
      }
      
      // 子串相同直接返回
      if (s1.substr(i, len) == s2.substr(j, len)) {
              memo[i][j][len] = 1;
              return true;
      }
      
      // 枚举切分点
      for (int k = 1; k < len; k++) {
          // 只要有一个切分点(无论交换与否)满足扰乱字符串，就直接返回
          // 1. 不交换: s1左对s2左，s1右对s2右
          if (dfs(s1, s2, i, j, k, memo) && dfs(s1, s2, i + k, j + k, len - k, memo)) {
              memo[i][j][len] = 1;
              return true;
          }
          
          // 2.交换: s1左对s2右，s1右对s2左
          if (dfs(s1, s2, i, j + len - k, k, memo) && dfs(s1, s2, i + k, j, len - k, memo)) {
              memo[i][j][len] = 1;
              return true;
          }
      }
      
      memo[i][j][len] = -1;
      return false;
  }
  
  bool isScramble(string s1, string s2) {
  	if (s1 == s2) return true;
      if (s1.length() != s2.length()) return false;
      int n = s1.length();
      
      // 为什么记忆数组用int而不是bool，因为需要表示"未计算"、"是"、"不是"三种状态
      // 注意len的范围为[0...n]，数组第三维大小为n+1
      vector<vector<vector<int>>> memo(n, vector<vector<int>>(n, vector<int>(n + 1, 0)));
      return dfs(s1, s2, 0, 0, n, memo);
  }
  ```

  下面看动态规划；

  还是要先明确依赖关系，以找出填表顺序和初始状态；

  观察递归过程可知，$dfs(i,j.len)$ 依赖于 $dfs(i,j,k),dfs(i+k,j+k,len-k),dfs(i,j+len-k,k),dfs(i+k,j,len-k)$，其中 $k\in[1,len-1]$，也就是说，被依赖子状态的第三个维度无论是 $k<len$ 还是 $len-k<len$；

  这意味着，沿着 $z$ 轴进行逐层填表时，当前层 (第 $len$ 层) 只依赖于下面 $1...len-1$ 层，而同层之间没有依赖关系；

  那么填表顺序就是：沿 $z$ 轴向上逐层填表，同层随意；

  需要初始化第 $1$ 层 (第 $0$ 层是未定义的)；

  这里就没办法采取滚动数组的空间压缩方式了；

  ```cpp
  bool isScramble(string s1, string s2) {
      if (s1 == s2) return true;
      if (s1.length() != s2.length()) return false;
      int n = s1.length();
      vector<vector<vector<bool>>> dp(n, vector<vector<bool>>(n, vector<bool>(n + 1, 0)));
      
      for (int i = 0; i < n; i++) {
          for (int j = 0; j < n; j++) {
              dp[i][j][1] = (s1[i] == s2[j]);
          }
      }
      
      for (int len = 2; len <= n; len++) {
          // 注意i和j变量的范围，第一次写根本没发现....
          for (int i = 0; i <= n - len; i++) {
              for (int j = 0; j <= n - len; j++) {
                  // 枚举分割点
                  for (int k = 1; k < len; k++) {
                      // 还是一样，有一个满足直接break去填下一个
                      // 情况 1: 不交换
                      if (dp[i][j][k] && dp[i + k][j + k][len - k]) {
                          dp[i][j][len] = true;
                          break;
                      }
                      // 情况 2: 交换
                      if (dp[i][j + len - k][k] && dp[i + k][j][len - k]) {
                          dp[i][j][len] = true;
                          break;
                      }
                  }
              }
          }
      }
      
      return dp[0][0][n];
  }
  ```

##### Problem 25 - Kadane Algorithm

- 题目：[LeetCode - 53. 最大子数组和](https://leetcode.cn/problems/maximum-subarray)；

  设 $dp[i]$：以索引 $i$ 结尾的连续子数组的最大累加和；则对于位置 $i$ 的元素 $nums[i]$：

  1. 延续前面的子数组：将 $nums[i]$ 追加到以 $nums[i - 1]$ 结尾的最大子数组后面，此时累加和为 $dp[i - 1] + nums[i]$；

     > 思考：为什么不用延续前面的部分子数组？

  2. 另起新的子数组：仅由 $nums[i]$ 本身构成一个长度为 1 的子数组，此时累加和为 $nums[i]$；

  则：$dp[i] = \max(dp[i - 1] + nums[i], nums[i])$；

  ```cpp
  int maxSubArray(vector<int>& nums) {
  	int n = nums.size();
      vector<int> dp(n, 0);
      dp[0] = nums[0];
      
      int ans = dp[0];
      for (int i = 1; i < n; i++) {
          dp[i] = max(dp[i - 1] + nums[i], nums[i]);
          ans = max(ans, dp[i]);
      }
      
      return ans;
  }
  ```

  ```cpp

      int ans = dp;
      for (int i = 1; i < nums.size(); i++) {
  ```

  如果不要求返回最大累加和，而是返回具有最大累加和的连续子数组的[left,right]索引呢？

  ```cpp

  pair<int, int> maxSubArray(const std::vector<int>& nums) {
      int dp = nums[0];
      int ans = nums[0];
      
      int current_start = 0;
      int best_left = 0;
      int best_right = 0;
      
      for (int i = 1; i < nums.size(); i++) {
          if (dp > 0) {
              dp += nums[i];
          } else {
              dp = nums[i];
              current_start = i; // 丢弃前面的负贡献，从 i 重新开始
          }
          
          if (ans < dp) {
              ans = dp;
              best_left = current_start;
              best_right = i;
          }
      }
      
      return {best_left, best_right};
  }
  ```

##### Problem 26

- 题目：[LeetCode - 198. 打家劫舍](https://leetcode.cn/problems/house-robber)；

  设 $dp[i]$ 表示在前 $i + 1$ 个元素 (索引 $0$ 到 $i$) 中，按约束条件能获取的最大值 (即不选相邻元素)；

  则来到第 $i$ 个位置有两种选择：

  - 不选 $nums[i]$：最大收益等于前 $i-1$ 个元素的最大值，即 $dp[i - 1]$；
  - 选 $nums[i]$：则不能选 $nums[i-1]$，最大收益等于前 $i-2$ 个元素的最大值加上当前值，即 $dp[i - 2] + nums[i]$ (题目保证 $nums[i]\ge0$)；

  取两种决策的最大值：$dp[i] = \max(dp[i - 1], dp[i - 2] + nums[i])$；

  ```cpp
  int rob(vector<int>& nums) {
  	int n = nums.size();
      if (n == 1) {
          return nums[0];
      }
      if (n == 2) {
          return max(nums[0], nums[1]);
      }
      
      vector<int> dp(n, 0);
      dp[0] = nums[0];
      dp[1] = max(nums[0], nums[1]);
      
      for (int i = 2; i < n; i++) {
          dp[i] = max(dp[i - 1], dp[i - 2] + nums[i]);
      }
      
      return dp[n - 1];
  }
  ```

  ```cpp
  // 空间压缩
  int rob(vector<int>& nums) {
      // 相当于给所有数组前面补两个零 (因为nums[i]>=0)
      int prev2 = 0; // 对应 dp[i-2]
      int prev1 = 0; // 对应 dp[i-1]
      
      for (int x : nums) {
          int curr = max(prev1, prev2 + x);
          prev2 = prev1;
          prev1 = curr;
      }
      
      return prev1;
  }
  ```

##### Problem 27

- 题目：[LeetCode - 918.环形子数组的最大和](https://leetcode.cn/problems/maximum-sum-circular-subarray)；

  注意不能将数组拷贝一份再套用 Kadane 算法，有可能会重复使用某些元素； 

  环形子数组最大和分两种情况：

  1. **不跨越边界**：即普通的全局最大子数组和 $\text{max\_sum}$；

  2. **跨越边界 (首尾相接)**：剩余未被选中的中间部分必须是连续的**最小子数组和** $\text{min\_sum}$；因此跨越边界的最大和等于 $\text{total\_sum} - \text{min\_sum}$；

     $\implies ans = \max(\text{max\_sum}, \text{total\_sum} - \text{min\_sum})$

  ```cpp
  int maxSubarraySumCircular(vector<int>& nums) {
  	int totalSum = 0;
      int curMax = 0, maxSum = nums[0];
      int curMin = 0, minSum = nums[0];
      
      for (int num : nums) {
          curMax = max(num, num + curMax);
          maxSum = max(curMax, maxSum);
          
          curMin = min(num, num + curMin);
          minSum = min(curMin, minSum);
          
          totalSum += num;
      }
      
      // 这里必须留意一种情况：数组全为负数
      // 此时minSum == totalSum, 而maxSum == 最大的nums[i]
      // 那显然max(maxSum, totalSum - minSum) = 0, 相当于最大子数组什么都没有，不选
      // 此题不允许这样选择，所以此时直接返回maxSum
      return minSum == totalSum ? maxSum : max(maxSum, totalSum - minSum);
  }
  ```

  > 跨边界子数组最大和不方便计算 $\to$ 计算中间的子数组最小和；
  >
  > 子数组最小和递推逻辑和最大和一致；
  >
  > 全负情况特判，防止 minSum == totalSum 时逻辑上 子数组最大和 选取到了空数组；

##### Problem 28

- 题目：[LeetCode - 213. 打家劫舍II](https://leetcode.cn/problems/house-robber-ii)；

  与 **Problem 2** 的区别在于需要考虑 $nums[0]$ 和 $nums[n-1]$ 是相邻的；

  那么可以分两种情况：

  1. 偷  $nums[0]$，不能偷 $nums[n-1]$；
  2. 不偷 $nums[0]$，可偷可不偷 $nums[n-1]$；

  $\implies ans = max\{nums[0]+rob(1,n-2),rob(1,n-1)\}$

  ```cpp
  int rob(vector<int>& nums) {
      int n = nums.size();
      if (n == 1) return nums[0];
  
      return max(nums[0] + helper(nums, 2, n - 2), helper(nums, 1, n - 1));
  }
  
  // 和Problem 2空间压缩版一致
  int helper(const vector<int>& nums, int left, int right) {
      int prev1 = 0;
      int prev2 = 0;
  
      for (int i = left; i <= right; ++i) {
          int curr = max(prev1, nums[i] + prev2);
          prev2 = prev1;
          prev1 = curr;
      }
  
      return prev1;
  }
  ```

  > 偷 $nums[0]$ 后，既不能偷 $nums[n-1]$，也不能偷 $nums[1]$，所以是 $nums[0]+helper(2,n-2)$；

  > 注意 $helper$ 有参数 $n-2$ ，所以要特判 $n==1$；
  >
  > $n==2$ 不用特判，$helper(2,0)$ 直接返回0，$helper(1,1)$ 返回 $nums[1]$，最终程序返回 $max(nums[0],nums[1])$；

  > 特判 $n==2$ 也是 `if (n === 2) return max(nums[0], nums[1]);`

##### Problem 29

- 题目：[LeetCode - 2560. 打家劫舍IV](https://leetcode.cn/problems/house-robber-iv)；

  题意简述：给定数组 $nums$，从中选择一个长度至少为 $k$ 的子序列 $A$，要求 $A$ 中没有任何元素在 $nums$ 中是相邻的；最小化 $max(A)$；(题目保证子序列 $A$ 存在) ($nums[i]\ge1$) ​

  思路：选择阈值 $X$，将满足 $nums[i]\le X$ 的元素选入子序列 $A$；如果在这样的子序列 $A$ 中能选出至少 $k$ 个在原数组中不相邻元素，说明 $X$ 可能取更小值；如果选不出来这样的 $k$ 个元素，那么必须增大 $X$；

  枚举 $min(nums)\le X\le max(nums)\implies$ 二分搜索；

  ```cpp
  int minCapability(vector<int>& nums, int k) {
  	int n = nums.size(), left = nums[0], right = nums[0];
      // 二分搜索范围
      for (int i = 0; i < n; i++) {
          left = min(left, nums[i]);
          right = max(right, nums[i]);
      }
      
      int ans = 0;
      while (left <= right) {
          int mid = left + (right - left) / 2;
          
          if (helper(nums, mid, k)) {
              ans = mid;
              right = mid - 1;
          } else {
              left = mid + 1;
          }
      }
      
      return ans;

  bool helper(const vector<int>& nums, int X, int k) {
      int n = nums.size();

      vector<int> dp(n + 1, 0);
  
      dp[0] = 0;
      dp[1] = (nums[0] <= X) ? 1 : 0;
  
      for (int i = 2; i <= n; i++) {
          if (nums[i - 1] <= X) {
              // 第i间房子(下标i-1)<=X时有两种选择，要么选要么不选
              dp[i] = max(dp[i - 1], dp[i - 2] + 1);
          } else {
              // 只能不选第i间房子
              dp[i] = dp[i - 1];
          }
  
          if (dp[i] >= k) return true; // 提前剪枝
      }
  
      return dp[n] >= k;
  }
  ```

##### Problem 30

- 题目：[LeetCode - 面试题17.24. 最大子矩阵](https://leetcode.cn/problems/max-submatrix-lcci)；

  假设要选取的子矩阵必须跨越第 $r1$ 行到第 $r2$ 行，思路是将这几行按列累加到一个一维数组上，然后执行 $Kadane$ 算法即可；

  所以枚举 $r1$ 与 $r2$ 即可；

  ```cpp
  vector<int> getMaxMatrix(vector<vector<int>>& matrix) {
      int m = matrix.size(), n = matrix[0].size();
      long long maxSum = LLONG_MIN;
      vector<int> ans(4, 0);
  
      // 枚举上边界
      for (int r1 = 0; r1 < m; r1++) {
          vector<long long> rowsSum(n, 0); // 刷新列压缩数据
          // 枚举下边界
          for (int r2 = r1; r2 < m; r2++) {
              // 准备进行 kadane 所需的变量
              long long currentSum = 0;
              int startCol = 0;
              
              for (int c = 0; c < n; c++) {
                  rowsSum[c] += matrix[r2][c]; // 压缩列数据 (累加当前行即可)
                  
                  // 一边压缩一边 kadane
                  if (currentSum > 0) {
                      currentSum += rowsSum[c];
                  } else {
                      currentSum = rowsSum[c];
                      startCol = c;
                  }
                  
                  if (currentSum > maxSum) {
                      // 别忘了刷新旧数据
                      maxSum = currentSum;
                      ans = {r1, startCol, r2, c};
                  }
              }
          }
      }
  
      return ans;
  }
  ```

  > 固定上边界和下边界后的矩阵选择实际上是一维的；
  >
  > 压缩列数据时，不要对于每个 $(r1,r2)$ 都执行一个双层  $for$ 的加法，只需在终止行递增时将当前终止行的数据累加到之前的结果上，然后起始行更新时刷新列压缩数据即可；

##### Problem 32

- 题目：给定数组 $nums$ ($nums[i]\ge0$)，从中挑选出累加和能被 $7$ 整除的**子序列**，返回最大的累加和；

  朴素的递归思路：每个 $nums[i]$ 要么选要么不选，那么直接遍历状态空间，将能被 $7$ 整除的子序列和与全局最大累加和比较并更新：

  ```cpp
  int ans = 0;
  
  // u: 当前考察到第 u 个数
  // current_sum: 当前已挑选元素的累加和
  void dfs(int u, int current_sum, const vector<int>& nums) {
      if (u == nums.size()) { // 触底
          if (current_sum % 7 == 0) {
              ans = max(ans, current_sum); // 与全局结果比较并更新
          }
          return;
      }
      // 决策 1: 不选 nums[u]
      dfs(u + 1, current_sum, nums);
      
      // 决策 2: 选 nums[u]
      dfs(u + 1, current_sum + nums[u], nums);
  }
  ```

  此方案存在的问题：

  - 时间复杂度为 $O(2^n)$，因为每个叶子节点都遍历了；
  - 难以记忆化，因为 $current\_sum$ 会不断增大；

  改进：压缩状态空间，只关心前 $i$ 个元素中模 $7$ 等于 $j$ 的子序列的最大累加和；状态空间被大大压缩 ($dp[0\sim n][0\sim6]$)，最后返回 $dp[n][0]$ 即可；

  不过值得注意的是，可能存在余数不可达的情况；

  递推与填表逻辑就不赘述了；

  ```cpp
  int maxSumDivisibleBy7(vector<int>& nums) {
      int n = nums.size();
  	vector<vector<int>> dp(n + 1, vector<int>(7, -1)); // -1表示该余数不可达
      
      dp[0][0] = 0; // 注意dp[0][1~6]要保持为-1，因为不可达
      for (int i = 1; i <= n; i++) {
          for (int j = 0; j < 7; j++) {
  			dp[i][j] = dp[i - 1][j]; // 1. 不选择当前元素
              
              // (prev_j + nums[i]) % 7 == j
              int prev_j = (j - nums[i - 1] % 7 + 7) % 7; // 理解prev_j的计算逻辑
              if (dp[i - 1][prev_j] != -1) { // 2. 在之前的余数可达的前提下选择
                  dp[i][j] = max(dp[i][j], dp[i - 1][prev_j] + nums[i - 1]); 
                  // 注意别忘了加上 nums[i - 1]
              }
          }
      }
      
      return dp[n][0];
  }
  ```

  > 仔细品味如何压缩的状态空间；同时要注意存在余数不可达的情况；
  >
  > 另外 $prev\_j$ 的计算逻辑看不懂直接询问LLM；
  >
  > 提醒一句哨兵 $dp[0]$ 中 $dp[0][0]=0$，而 $dp[0][1\sim6]$ 均为 $-1$；

##### Problem 33

- 魔法卷轴问题：给定整数数组 $nums$，一个魔法卷轴可以将数组中连续的一段全部变成 $0$，共有两张魔法卷轴，使不使用随意；如何使数组的累加和最大？返回最终的最大累加和；

  可以证明，该过程 $\iff$ 将数组一分为二，分别在两个子数组上运行 $minKadane$ 算法；

  设数组初始总和为 $S = \sum nums[i]$；选出至多两个**互不重叠**的连续子数组，使其和尽量小，且必须小于 $0$，最终最大和为 $S - min_1 - min_2$；

  - 注：两个子数组重叠或者相邻相当于只使用一张卷轴；如何最小和子数组大于零，那么不使用这张卷轴；

  所以可将数组分成两部分处理，左右各求最小累加和，然后枚举分割点；

  ```cpp
  int magicScroll(vector<int>& nums) {
      int n = nums.size();
      if (n == 0) return 0;
      if (n == 1) return max(0, nums[0]);
  
      // 1. 计算初始总和 S
      int totalSum = 0;
      for (int i = 0; i < n; i++) {
          totalSum += nums[i];
      }
  
      // 2. 初始最佳答案为原总和（一张卷轴都不用）
      int maxAns = totalSum;
  
      // 3. 枚举分割点 i：左半部分 [0, i]，右半部分 [i + 1, n - 1]
      for (int i = 0; i < n - 1; i++) {
          // 如果最小和大于 0，说明抹去反而会减小总和，因此不使用该卷轴（置为 0）
          int min1 = min(0, minKadane(nums, 0, i));
          int min2 = min(0, minKadane(nums, i + 1, n - 1));
  
          // 统一基于初始总和 totalSum 计算抹去负数后的收益
          maxAns = max(maxAns, totalSum - min1 - min2);
      }
  
      return maxAns;
  }
  
  // 辅助函数：求子数组 [left, right] 内的最小连续子数组和
  int minKadane(const vector<int>& nums, int left, int right) {
      if (left == right) return nums[left];
      
      int curSum = nums[left];
      int minVal = nums[left];
      
      for (int i = left + 1; i <= right; i++) {
          curSum = min(nums[i], curSum + nums[i]);
          minVal = min(minVal, curSum);
      }
      
      return minVal;
  }
  ```

  - 分析：对于每个分割点都要调用两次 $minKadane$，时间复杂度会来到 $O(2n^2)$；

  改进：前后缀预处理；

  设`leftMin[i]`是`nums[0...i]`上的最小子数组和；正向遍历时，维护以当前位置结尾的最小子数组和`curMin = min(nums[i], curMIn + nums[i])`；则`leftMin[i] = min(leftMin[i - 1], curMin)`；

  设`rightMin[i]`是`nums[i...n-1]`上的最小子数组和；逻辑一致；

  前面的代码是只会用 $kadane$，后面的代码才是理解了 $kadane$；

  ```cpp
  int magicScroll(vector<int>& nums) {
      int n = nums.size();
      if (n == 0) return 0;
      if (n == 1) return max(0, nums[0]);
      
      // 预处理前缀
      vector<int> leftMin(n, 0); // leftMin[i]代表nums[0...i]上最小子数组和，如果大于0要置0
      int curMin = 0; // 以当前位置结尾的最小子数组和
      int minSofar = 0;
      for (int i = 0; i < n; i++) {
          curMin = min(nums[i], curMin + nums[i]);
          minSofar = min(curMin, minSofar);
          leftMin[i] = min(0, minSofar); // 新增：如果大于零要变成零，相当于不使用卷轴
      }
      
      // 预处理后缀
      vector<int> rightMin(n, 0);
    	curMin = 0;
      minSofar = 0;
      for (int i = n - 1; i >= 0; i--) {
  		curMin = min(nums[i], curMin + nums[i]);
          minSofar = min(curMin, minSofar);
          rightMin[i] = min(0, minSofar);
      }
      
      int totalSum = 0;
      for (int i = 0; i < n; i++) {
          totalSum += nums[i];
      }
      
      int ans = totalSum; // 一次卷轴都不使用
      for (int i = 1; i < n; i++) { // 枚举分割点
      	ans = max(ans, totalSum - leftMin[i - 1] - rightMin[i]);    
      }
      
      return ans;
  }
  ```

  >

##### Problem 34

- 题目：[LeetCode - 689. 三个无重叠子数组的最大和](https://leetcode.cn/problems/maximum-sum-of-3-non-overlapping-subarrays)；

  由于三个子数组的长度确定且均为 $k$，那么第二个子数组的起点 $j$ 的取值范围为 $[k, n - 2k]$​，当 $j$ 确定时：第一个子数组的合法起点范围是 $[0, j - k]$，第三个子数组的合法起点范围是 $[j + k, n - k]$；

  只需遍历 $k$，然后在前面和后面运行 $kadane$ 算法即可，于是又可进行前后缀预处理；

  ```cpp
  vector<int> maxSumOfThreeSubarrays(vector<int>& nums, int k) {
  	int n = nums.size();
      
  	// 1. 前后缀预处理的辅助数组sums，sum[i]存储[i..i+k-1]的k个元素和
      vector<int> sums(n - k + 1, 0); // 注意sums长度
      for (int i = 0 ; i < k; i++) {  // 先计算sums[0]
          sums[0] += nums[i];
      }
      // 注意利用滚动法计算sums[i]，不要双重循环
      for (int i = 1; i < n - k + 1; i++) {
          sums[i] = sums[i - 1] + nums[k + i - 1] - nums[i - 1];
      }
      
      // 2. 前后缀预处理
      // 由于本题要求返回数组的最小字典序起始位置，所以：
      // left[i]表示[0..i]上最大sums[i]的起始位置(就是i)(有多个则取最小字典序)
      // right[i]表示[i..n-k]上最大sums[i]的起始位置(有多个取最下字典序)
      vector<int> left(n - k + 1, 0);
      int bestLeft = 0;
      for (int i = 0; i < n - k + 1; i++) {
          if (sums[i] > sums[bestLeft]) { // 这里不要写成>=，否则不是最小而是最大字典序
  			bestLeft = i;
          }
          left[i] = bestLeft;
      }
      
      vector<int> right(n - k + 1, 0);
      int bestRight = n - k;
      // 注意要倒序，因为求的是后缀区间上的最大
      for (int i = n - k; i >= 0; i--) {
  		if (sums[i] >= sums[bestRight]) {// 注意这里又要写成>=才是最小字典序
          	bestRight = i;
          }
          right[i] = bestRight;
      }
      
      // 3. 枚举第二个子数组的起始点j
      vector<int> ans(3 ,0);
      int totalSum = 0;
      for (int j = k; j < n - 2 * k; j++) {
          // 第一个子数组是[0..j-k]上最大的sums，也就是sums[left[j-k]]
          // 第三个子数组是[j+k..n-k]上最大的sums，也就是sums[right[j+k]]
          // 第二个子数组就是sums[j]
          int curSum = sums[left[j - k]] + sums[j] + sums[right[j + k]];
          if (curSum > totalSum) { // 一样的，不要写为>=，否则不是最小字典序
  			totalSum = curSum;
              ans = {left[j - k], j, right[j + k]};
          }
      }
      
      return ans;
  }
  ```

  > 问题的转化，原问题 $\to$ 枚举第二个数组起始点并在两边运行 $kadane$ $\to$ 前后缀预处理，和上一题很相似；
  >
  > 不要死板的定义前后缀，比如这里就变成了起始位置 $i$；

##### Problem 35

- 将一个数组的中间的某一段翻转后能得到的最大子数组累加和是多少？

  首先随便翻转一段子数组，如果最大子数组和翻转子数组没有交集或者最大子数组包含了翻转子数组或者翻转子数组包含了最大子数组，那么相当于没有翻转，等价于在原数组上求解最大子数组；

  然后如果最大子数组和翻转子数组是相交关系，其实相当于在原数组上求两个不相交子数组最大累加和；

  所以综上，最终只用求两个不相交子数组的最大累加和 (因为相邻相当于求一个)；

  那么代码写法也非常清晰了，枚举分界线，两边运行 $kadane$，然后又可以前后缀预处理；

  ```cpp
  int maxSubarraySumAfterOneReversalPrefixSuffix(const vector<int>& nums) {
      int n = nums.size();
      if (n == 0) return 0;
      if (n == 1) return nums[0];
      
      // 1. 前后缀预处理
      // left[i]表示[0..i]上最大子数组和
      vector<int> left(n, 0);
      left[0] = nums[0];
      int curMax = nums[0];
      int singleMax = nums[0]; // 这个变量有什么用？
      for (int i = 1; i < n; i++) {
  		curMax = max(nums[i], curMax + nums[i]);
          left[i] = max(left[i - 1], curMax);
          singleMax = max(singleMax, curMax);
      }
      
      // right[i]表示[i..n-1]上最大子数组和
      vector<int> right(n, 0);
      right[n - 1] = nums[n - 1];
      curMax = nums[n - 1];
      // 注意后缀区间要反序
      for (int i = n - 2; i >= 0; i--) {
  		curMax = max(nums[i], curMax + nums[i]);
          right[i] = max(right[i + 1], curMax);
      }
      
      // 2. 枚举分割点
      int ans = singleMax;
      for (int i = 0; i < n - 1; i++) {
          ans = max(ans, left[i] + right[i + 1]);
      }
      
      return ans;
  }
  ```

  > 这里重点说一下 $singleMax$ 变量的用意：
  >
  > 思路中说了，分割点边分别求最大子数组也包含了整体单独求一个子数组，但是有个致命问题是，我们求的分割点两边的子数组都是非空的，这会漏掉一边空一边不空的情况；
  >
  > 比如 $nums=[-3,-1,-2]$，正确答案是 $-1$，但是我们求出来会变成 $-3$ (分割点 $i=1$，左边 $-1$，右边 $-2$)；
  >
  > 所以需要维护一个全局最大子数组和 $singleMax$，并且这个变量不用单独跑一次 $kadane$，处理前缀 / 后缀时顺便就做了，所以上述代码中填 $right$ 数组时计算 $singleMax$ 也是可以的；

##### Problem 20 - 放到动态规划中做反例

- 题目：[洛谷 - P4799](https://www.luogu.com.cn/problem/P4799)；

  ```cpp
  using ll = long long;
  
  // 常规带状态DFS，但是main传参时必须把搜索树高度框死在N/2，不然状态爆炸
  void dfs(ll remain, const vector<ll>& prices, int index, int maxIndex, ll totalPrice, vector<ll>& sums) {
  	if (index >= maxIndex) {
  		sums.push_back(totalPrice);
  		return;
  	}
  
  	if (remain >= prices[index]) {
  		dfs(remain - prices[index], prices, index + 1, maxIndex, totalPrice + prices[index], sums);
  	}
  
  	dfs(remain, prices, index + 1, maxIndex, totalPrice, sums);
  
  	return;
  }
  
  int main() {
  	int N;
  	ll M;
  	if (!(cin >> N >> M)) return 0;
  
  	vector<ll> prices(N);
  	for (int i = 0; i < N; i++) {
  		cin >> prices[i];
  	}
  	
  	vector<ll> left, right;
      // dfs 设计的搜索范围是[index, maxIndex)
      // 所以left是[0, N/2), right是[N/2, N)
  	dfs(M, prices, 0, N / 2, 0, left);
  	dfs(M, prices, N / 2, N, 0, right);
  	sort(left.begin(), left.end());
  	sort(right.begin(), right.end());
  
  	ll ans = 0;
  	int j = static_cast<int>(right.size()) - 1;
  	for (ll totalPriceLeft : left) { // left 按升序遍历，可匹配的 right 范围只会缩小，因此 j 只向左移动
  		// 循环结束后，right[0..j] 都能与当前左侧方案配对；j == -1 时没有可配对方案
  		while (j >= 0 && right[j] > M - totalPriceLeft) {
  			--j;
  		}
  		ans += j + 1;
  	}
  
  	cout << ans << "\n";
  	return 0;
  }
  ```

##### Problem 21 - DP反例

- 题目：[LeetCode - 1755. 最接近目标值的子序列和](https://leetcode.cn/problems/closest-subsequence-sum)；

  ```cpp
  class Solution {
  private:
  	void dfs(const vector<int>& nums, int index, const int& maxIndex, vector<long long>& ans, long long sum) {
  		if (index >= maxIndex) {
  			ans.push_back(sum);
  			return;
  		}
  
  		dfs(nums, index + 1, maxIndex, ans, sum + nums[index]);
  		dfs(nums, index + 1, maxIndex, ans, sum);
  
  		return;
  	}
  	
  public:
      int minAbsDifference(vector<int>& nums, int goal) {
  		int n = nums.size();
  
  		// 折半DFS
  		vector<long long> left, right;
  		dfs(nums, 0, n / 2, left, 0);
  		dfs(nums, n / 2, n, right, 0);
  
  		sort(left.begin(), left.end());
  		sort(right.begin(), right.end());
  
  		// 双指针找答案
  		int i = 0, j = right.size() - 1;
  		long long ans = 1e9;
  		while (i < left.size() && j > -1) {
  			long long sum = left[i] + right[j];
  			ans = min(ans, llabs(sum - goal));
  
  			if (sum == goal) break;
  			if (sum > goal) { // sum大了需要缩小
  				j--;
  			} else { // sum小了需要增大
  				i++;
  			}
  		}
  
  		return ans;
      }
  };
  ```

  ```cpp
  class Solution {
  private:
  	void dfs(const vector<int>& nums, int index, const int& maxIndex, vector<long long>& ans, long long sum) {
  		if (index >= maxIndex) {
  			ans.push_back(sum);
  			return;
  		}
  
  		int snc = 1;
  		while (index + snc < maxIndex && nums[index + snc] == nums[index]) {
  			++snc;
  		}
  
  		for (int i = 0; i <= snc; ++i) { // 将二叉树变为多叉树，降低递归深度
  			dfs(nums, index + snc, maxIndex, ans, sum + 1LL * i * nums[index]);
  		}
  		return;
  }
  	
  public:
      int minAbsDifference(vector<int>& nums, int goal) {
  		int n = nums.size();
  		
  		// 常数优化1，比较goal与最大和 / 最小和
  		int maxSum = 0, minSum = 0;
  		for (int i = 0; i < n; i++) {
  			nums[i] > 0 ? maxSum += nums[i] : minSum += nums[i];
  		}
  		if (maxSum < goal) { return goal - maxSum; }
  		if (minSum > goal) { return minSum - goal; }
  
  		// 折半DFS
  		sort(nums.begin(), nums.end()); // 常数优化2，给nums排序便于dfs多叉展开
  		vector<long long> left, right;
  		dfs(nums, 0, n / 2, left, 0);
  		dfs(nums, n / 2, n, right, 0);
  
  		sort(left.begin(), left.end());
  		sort(right.begin(), right.end());
  
  		// 双指针找答案
  		int i = 0, j = right.size() - 1;
  		long long ans = 1e9;
  		while (i < left.size() && j > -1) {
  			long long sum = left[i] + right[j];
  			ans = min(ans, llabs(sum - goal));
  
  			if (sum == goal) break;
  			if (sum > goal) {
  				j--;
  			} else {
  				i++;
  			}
  		}
  
  		return ans;
      }
  };
  ```
